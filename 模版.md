## 树状数组
```cpp
struct bit {
    int n;
    vector<ll> tree;
    bit(int n1): n(n1), tree(n + 2, 0) {}
    void add(int i, ll delta) {
        if (i <= 0) return;
        for (; i <= n; i += i & -i)
            tree[i] += delta;
    }
    ll query(int i) {
        ll sum = 0;
        for (; i > 0; i -= i & -i)
            sum += tree[i];
        return sum;
    }
};
```

## 最大公因数 GCD
```cpp
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}
```

## 最小公倍数 LCM
```cpp
int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}
```

## 快速幂
```cpp
ll fast_pow(ll a, ll b) {
    ll res = 1;
    while (b > 0) {
        if (b & 1) res = (res * a) % MOD;
        a = (a * a) % MOD;
        b >>= 1;
    }
    return res;
}
```

## 并查集
```cpp
class UnionFold {
    private:
        int count;
        vector<int> parent;
        vector<int> size; 
    public:
        UnionFold(int n) : count(n), parent(n), size(n, 1) {
            for (int i = 0; i< n; i++) {
                parent[i]=i;
            }
        }
        void unionSets(int p, int q){
            int rootP = find(p);int rootQ = find(q);
            if (rootP == rootQ) return;
            if (size[rootP] > size[rootQ]) {
                parent[rootQ] = rootP; size[rootP] += size[rootQ];
            } else {
                parent[rootP] = rootQ; size[rootQ] += size[rootP];
            }
            count--;
        }
        bool connected(int p, int q) { return find(q) == find(p); }
        int find(int x) {
            if (x == parent[x]) return x;
            return parent[x] = find(parent[x]);
        }
        int Count() { return count; }
};
```

## 最短路 Dijkstra
```cpp
const ll INF = 4e18;
struct state {
    int node;
    ll dis;
    state(int n, ll d): node(n), dis(d) {}
    bool operator<(const state& other) const {
        return dis > other.dis;
    }
};
vector<ll> Dijkstra(int start) {
    vector<ll> dis(n + 1, INF);
    vector<int> parent(n + 1, -1);
    priority_queue<state> q;
    q.push(state(start, 0LL));
    dis[start] = 0;
    while (!q.empty()) {
        auto [u, w] = q.top(); q.pop();
        if (dis[u] != w) continue;
        for (auto& [v, w2]: mp[u]) {
            if (w + w2 < dis[v]) {
                dis[v] = w + w2;
                parent[v] = u;
                q.push({v, w + w2});
            }
        }
    }
    return dis;
}
```
