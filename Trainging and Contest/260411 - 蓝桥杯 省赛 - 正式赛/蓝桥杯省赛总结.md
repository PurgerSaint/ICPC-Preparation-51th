1. 试题 A: 魔法矩阵能量值

    模拟，O(n^2)遍历即可，样例里有提示，最终结果为基础能量 + 对角线加成 + 边界惩罚 + 中心奖励，后三个是均有的，非独立存在，使用double确保除于二存在小数

    暴力代码，测试样例均通过，输入13得到138966

    ```C++
    #include<bits/stdc++.h>
    using namespace std;
    #define ios ios::sync_with_stdio(false);cin.tie(0);
    using ll = long long;
    int main() {
    	ios;
    	int n;
    	cin >> n;
    	int c = n % 2 == 1 ? (n / 2 + 1) : -1;
    	double sum = 0;
    	for (int i = 1; i <= n; i++)
    		for (int j = 1; j <= n; j++) {
    			double t = i * j * (i + j);
    			if (i == j) t *= 2.0;
    			if (i == 1 || i == n || j == 1 || j == n) t /= 2.0;
    			if (i == c && j == c) t += 100;
    			sum += t;
    		}
    	cout << sum;
    	return 0;
    }
    ```

1. 试题 E: 纯粹魔药

    给与多个魔力值，可以对其中一个魔力值进行转换，要求最终所有魔力值乘积为质数

    【重点】如果要求乘积为质数，则必定这些魔力值只剩下一个质数或多个1，多个质数、多个非质数（无论是否转换后最简）的乘积必定不可能是质数

    那么就可以直接排除以下情况：

    1. 原序列中只有一个数出现（例如多个1）

    2. 原序列中出现多个非1数（两个或两个以上）

    然后就对只出现一次的那个非1数，若是质数直接返回，若不是就继续化简，化简到最终看是否为质数返回即可，使用了埃氏筛

    ```C++
    #include<bits/stdc++.h>
    using namespace std;
    #define ios ios::sync_with_stdio(false);cin.tie(0);
    using ll = long long;
    unordered_map<ll, set<ll>> mp;
    vector<bool> flag;
    void expand(ll val) {
    	for (ll i = flag.size(); i <= val; i++)
    		flag.push_back(true);
    	for (int i = 2; i <= val; i++) {
    		if (flag[i]) {
    			for (int j = 2; (i * j) <= val; j++) {
    				if (!flag[i * j]) continue;
    				flag[(i * j)] = false;
    				mp[i * j].insert(i);
    				mp[i * j].insert(j);
    			}
    		}
    	}
    }
    
    string solve() {
    	int m; cin >> m;
    	ll t = 0, bad = 0;
    	for (int i = 0; i < m; i++) {
    		ll val; cin >> val;
    		if (val != 1) {
    			if (t != 0 && val != t) bad = 1;
    			else t = val;
    		}
    	}
    	if (bad || t == 0) return "NO";
    	if (t + 1 >= flag.size()) expand(t);
    	flag[0] = false;
    	flag[1] = false;
    	if (flag[t]) return "YES";
    	else {
    		while (!flag[t] && mp.count(t)) {
    			t = mp[t].size() + 2;
    		}
    	}
    	
    	return flag[t] ? "YES" : "NO";
    }
    int main() {
    	ios;
    	int t; cin >> t;
    	while (t--) {
    		cout << solve() << (t != 0 ? "\n" : "");
    	}
    	return 0;
    }
    ```

1. 试题 F: 基态坍缩

    有L和Q两个人，L先手，每个人交替操作，但只能操作最后一个数

    可以将该数B转为0≤x<B中的任意一个数x，如果变为0则直接删除，倒数第二个数变为最后一个数

    第一眼看到题目，以为对B能有多个操作（减1，还是减2，亦或是有规律减），但都不是

    例如对于L，假设当前序列中数字个数为奇数，即代表操作次数（加上当前操作次数），按双方最优想法，即对于对手Q，必定要将数置为0，那么当前奇数操作次数，直接删除最后一个数（置为0），就能保证序列中最后一个数由我操作，那么接下来Q将会遇到空序列而失败；对于操作次数为偶数，我（L）必定要创造条件，使得我操作完像奇数操作次数一样，只剩下偶次，使得最后一次让我而来，那么此时对最后一个数减1还是减多少都无所谓，但不能直接清零，这里就按减1即可

    同理，那么就可以总结这俩的操作逻辑：

    1. 如果最后一个数大于1，对于L，若当前序列数量为偶数（即操作次数为偶数，算入当前），对最后一个数减一即可（因为无论减多少，Q都可以直接清零），若当前操作次数为奇数，直接清零即可，这样能保证L最优（最后一步为L）；对于Q，若操作次数为偶数，直接清零，若当前操作次数为奇数，则最后一数减一（使得最后一步为Q）

    2. 如果当前序列只剩一个数，无论多少，对于L和Q的最优选择都是直接清零，如果进行减，那么就是给对手创造机会（非最优选择）

    3. 如果最后一个数为1，正常走（清零）

    ```C++
    #include<bits/stdc++.h>
    using namespace std;
    #define ios ios::sync_with_stdio(false);cin.tie(0);
    using ll = long long;
    string solve() {
    	int n; cin >> n;
    	vector<ll> a(n);
    	for (ll& t: a) cin >> t;
    	ll cnt = 0;
    	while (!a.empty()) {
    		if (a.back() > 1) {
    			if (cnt % 2 == 0) {
    				// cout << "L";
    				if (a.size() > 1 && a.size() % 2 == 0) {
    					a.back() -= 1;
    					// cout << " step:1";
    				} else {
    					a.pop_back();
    					// cout << " step:2";
    				}
    			} else {
    				// cout << "Q";
    				if (a.size() > 1 && a.size() % 2 != 0) {
    					a.back() -= 1;
    					// cout << " step:1";
    				} else {
    					a.pop_back();
    					// cout << " step:2";
    				}
    			}
    			// cout << " " << a.size() << "\n";
    		} else {
    			a.pop_back();
    			// cout << (cnt % 2 == 0 ? "L" : "Q") << " end " << a.size() << "\n";
    		}
    		cnt++;
    	}
    	return cnt % 2 == 0 ? "Q" : "L";
    }
    int main() {
    	ios;
    	int t; cin >> t;
    	while (t--) {
    		cout << solve() << (t != 0 ? "\n" : "");
    	}
    	return 0;
    }
    ```

1. 试题 G: 读取指令

    给与一个可构造的数列，要求在每个数只能读取一次的情况下，最少读取多少次（可以连续读）使得读取总结果为W

    由于数列由C构建，那么如果**W非C的倍数**，可以直接否定

    逐步构建数列，同时创立前缀和，判断每个数是否为W或前缀和是否为W（无论连续读取多少，只要够，都是一次）

    【修补】由于前缀和默认升序，第一次通过二分查找（lower_bound）来寻找连续的（发现不行，对于样例2会出错），第二次用左右指针的滑动窗口，但最终都不如现在set直接存储查找方便

    然后再从后往前倒序查找，按贪心（选最大）来计算最终结果和读取次数

    ```C++
    #include<bits/stdc++.h>
    using namespace std;
    #define ios ios::sync_with_stdio(false);cin.tie(0);
    using ll = long long;
    ll solve() {
    	int n, c;
    	ll w;
    	cin >> n >> c >> w;
    	if (n == 1) {
    		return c == w ? 1 : 0;
    	}
    	if (w % c != 0) return -1;
    	vector<ll> a(n);
    	ll cnt = LLONG_MAX;
    	ll sum = 0;
    	set<ll> sum_hash;
    	for (int i = 1; i <= n; i++) {
    		a[i - 1] = i * c;
    		sum += a[i - 1];
    		
    		if (sum == w || a[i - 1] == w) {
    			cnt = 1;
    			break;
    		}
    		
    		if (sum_hash.count(sum - w)) {
    			cnt = 1;
    			break;
    		}
    		sum_hash.insert(sum);
    	}
    	if (cnt == 1) return 1;
    	
    	ll sum_back = 0, scnt = 0, last_i = -1;
    	for (int i = n - 1; i >= 0; i--) {
    		if (sum_back + a[i] > w) {
    			continue;
    		}
    		if (last_i == -1 || last_i - 1 > i) scnt++;
    		sum_back += a[i];
    		last_i = i;
    	}
    	if (sum_back == w) cnt = min(scnt, cnt);
    	
    	return cnt == LLONG_MAX ? -1 : cnt;
    }
    int main() {
    	ios;
    	int t; cin >> t;
    	while (t--) {
    		cout << solve() << (t != 0 ? "\n" : "");
    	}
    	return 0;
    }
    ```
