```cpp
1:2->3->4
2:5->6
3:7
4:7->8
5:
6:
7:8

存在一个这样的邻接表，其对应图将使用BFS和DFS进行访问。
对于深度优先搜索DFS，1->2->5->6（没有邻接节点，返回上一层。上一层5也没有，继续返回，直到第1层）->3->7->8->4。
对于广度优先搜索BFS，先访问完每一层，再逐层递进。
先访问根节点及其邻接节点：1->2->3->4
没有节点可供访问了，就从本次第一个邻接节点2继续：
->5->6
节点2的邻接节点也访问完了，就从节点3开始：
->7
继续节点4：
->（7访问过了，移除）->8
这样就完成了整个序列的访问。
故，对于DFS来说，由于其需要回溯，需使用递归；BFS则是队列（第一个邻接节点）。
```

洛谷P5318 **【深基18.例3】查找文献**

给与N个节点，M条有向边，问你从根节点开始，BFS和DFS的访问顺序是咋样的？

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
vector<vector<int>> image;
vector<bool> flag;
// 沿着一个方向走到头
void dfs(int i,vector<int>& res){
    res.push_back(i);   // 保存当前节点
    flag[i]=true;       // 记录已访问
    // 寻找下一个可以走的位置
    for(int& t:image[i])
        if(!flag[t])
            dfs(t,res);
}
// 逐层遍历
void bfs(int i,vector<int>& res){
    queue<int> q;
    q.push(i);
    flag[i]=true;

    while(!q.empty()){
        int t=q.front();
        res.push_back(t);
        q.pop();
        for(int& v:image[t]){
            if(!flag[v]){
                q.push(v);
                flag[v]=true;
            }
        }
    }
}
int main(){
    IOS;
    int n,m;cin>>n>>m;
    image = vector<vector<int>>(n+1);
    flag = vector<bool>(n+1,false);
    // x指向y
    for(int i=1;i<=m;i++){
        int x,y;cin>>x>>y;
        image[x].push_back(y);
    }
    // 优先看编号小的
    for(vector<int>& p:image)
        sort(p.begin(),p.end());

    auto print = [](vector<int>& res){
        int n=res.size();
        for(int i=0;i<n;i++)
            cout<<res[i]<<(i!=n-1?" ":"");
    };

    vector<int> res;
    dfs(1,res);
    print(res);
    cout<<"\n";
    res.clear();
    fill(flag.begin(),flag.end(),false);
    bfs(1,res);
    print(res);

    return 0;
}
```

洛谷P3366 【模版】最小生成树

给出一个无向图，求出最小生成树，如果该图不连通，则输出 `orz`。

> prim算法
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
vector<vector<pair<int,int>>> p;  //邻接表[邻居,权值]
vector<bool> flag;                //访问记录
vector<vector<int>> tree; //最终生成的最小树

// 从一个点开始,每次选择连接该点权值最小的边
// 然后从这个边到下个点,依旧选择该点权值最小的、对应点未被访问的边
// 直到逐步构建出一个最小生成树 
int prim(int start,int n){
    vector<bool> flag(n+1,false);   //记录是否被访问过
    // 用vector作为底层容器,以pair<int,int>作为数据结构的升序最小堆队列
    priority_queue<
        tuple<int,int,int>,             //{权值,起点,终点}
        vector<tuple<int,int,int>>,     //底层容器
        greater<tuple<int,int,int>>
    > pq;
    int sum=0;      //总和

    // 标记初始点已访问
    flag[start] = true;
    // 存储初始点的所有边
    for(auto& edge:p[start]){
        int v=edge.first,w=edge.second;
        pq.push(make_tuple(w,start,v));
    }

    while(!pq.empty()) {
        // 取得连接该点中最小权值的边
        auto t=pq.top();
        int w=get<0>(t),u=get<1>(t),v=get<2>(t);
        // 去除该边
        pq.pop();
        // 如果该边对应的点已访问,就跳过
        if(flag[v]) continue;
        // 否则标记该点已访问
        flag[v]=true;
        // 计入权值
        sum += w;
        // 加入树
        tree[u].push_back(v);
        tree[v].push_back(u);

        // 存入对应点的所有连接的边
        for(auto& edge:p[v]){
            int to=edge.first,w2=edge.second;
            if(!flag[to]) pq.push(make_tuple(w2,v,to));
        }
    }
    
    if(find(flag.begin()+1,flag.end(),false)!=flag.end()) return -1;
    
    return sum;
}

int main(){
    IOS;
    int n,m;cin>>n>>m;
    p = vector<vector<pair<int,int>>>(n+1);
    tree = vector<vector<int>>(n+1);
    // 最小权值所对应的边
    pair<int,int> x={0,INT_MAX};
    for(int i=0;i<m;i++){
        int a,b,c;cin>>a>>b>>c;
        p[a].push_back({b,c});
        p[b].push_back({a,c});
    }
    int sum=prim(1,n);
    if(sum==-1)	cout<<"orz";
    else cout<<sum;
    return 0;
}
```

> Kruskal算法
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
struct edge{
    int x,y,w;  //起点,终点,权值
    edge(int a,int b,int c):x(a),y(b),w(c) {}
};
vector<edge> E;             //无向图
vector<vector<int>> tree;   //最小生成树的邻接表
class UnionFold {
private:
    vector<int> parent;
    vector<int> size;

public:
    UnionFold(int n):parent(n+1),size(n+1,1) {
        for(int i=0;i<=n;i++)
            parent[i]=i;
    }

    int find(int x){
        while(x != parent[x]){
            parent[x]=parent[parent[x]];
            x=parent[x];
        }
        return x;
    }

    bool merge(int p,int q){
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP == rootQ) return false;

        if (size[rootP] > size[rootQ]){
            parent[rootQ] = rootP;
            size[rootP] += size[rootQ];
        } else {
            parent[rootP] = rootQ;
            size[rootQ] += size[rootP];
        }
        return true;
    }
};

// kruskal算法是每次选择最小的边
// 并且检查若加入该边是否会生成环
void kruskal(int n){
    // 排序使得每次拿取的都是最小的
    sort(E.begin(),E.end(),[](edge& a,edge& b){
        return a.w<b.w;
    });
    // 并查集防止生成的树有环
    UnionFold uf(n);
    // 权值总和和计数
    int sum = 0,cnt = 0;
    for(auto& e:E){
        // 合并失败就代表有连接
        if(uf.merge(e.x,e.y)){
            sum += e.w;
            tree[e.x].push_back(e.y);
            cnt++;
            if(cnt == n-1) break;
        }
    }
    // 如果不联通
    if(cnt<n-1) cout<<"orz";
    else cout<<sum;
}

int main(){
    IOS;
    int n,m;cin>>n>>m;
    tree = vector<vector<int>>(n+1);
    for(int i=0;i<m;i++){
        int a,b,c;cin>>a>>b>>c;
        E.push_back(edge(a,b,c));
    }
    kruskal(n);
    return 0;
}
```

力扣547.省份数量

有 `n` 个城市，其中一些彼此相连，另一些没有相连。如果城市 `a` 与城市 `b` 直接相连，且城市 `b` 与城市 `c` 直接相连，那么城市 `a` 与城市 `c` 间接相连。

**省份** 是一组直接或间接相连的城市，组内不含其他没有相连的城市。

给你一个 `n x n` 的矩阵 `isConnected` ，其中 `isConnected[i][j] = 1` 表示第 `i` 个城市和第 `j` 个城市直接相连，而 `isConnected[i][j] = 0` 表示二者不直接相连。

返回矩阵中 **省份** 的数量。

> DFS版本
> 

```cpp
class Solution {
public:
    vector<bool> flag;

    // 记录该省份中所有城市
    void dfs(int i,vector<vector<int>>& isConnected){
        flag[i]=true;
        int n=isConnected.size();
        for(int j=0;j<n;j++)
            if(isConnected[i][j]==1&&!flag[j])
                dfs(j,isConnected);
    }
    int findCircleNum(vector<vector<int>>& isConnected) {
        int n=isConnected.size();
        flag = vector<bool>(n);
        int cnt = 0;
        for(int i = 0;i < n;i++){
            if(!flag[i]){
                dfs(i,isConnected);
                cnt++;
            }
        }
        return cnt;
    }
};
```
