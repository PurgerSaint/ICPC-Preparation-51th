```cpp
/* 并查集完全版 */
class UnionFold {
private:
    /* 连通分量[不同集合]个数 */
    int count;              //连通分量个数
    /* 每个节点的上一节点[祖先] */
    vector<int> parent;     //祖先
    /* 以每个节点为根节点的树大小 */
    vector<int> size;       //大小

public:
    // 初始化并查集
    UnionFold(int n) : count(n), parent(n), size(n, 1) {
        for (int i = 0; i< n; i++) {
            parent[i]=i;
        }
    }

    // 将 p 和 q 连通
    void unionSets(int p, int q){
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP == rootQ) return;

        // 以每个节点为根节点的树大小作为权重
        // 将小树连接到大树下面
        if (size[rootP] > size[rootQ]) {
            parent[rootQ] = rootP;
            size[rootP] += size[rootQ];
        } else {
            parent[rootP] = rootQ;
            size[rootQ] += size[rootP];
        }
        count--;
    }

    // 判断 p 和 q 是否互相连通[在同一集合内]
    bool connected(int p, int q) {
        return find(q) == find(p);
    }

    // 寻找 x 所在树的根节点
    int find(int x) {
        //寻找x的祖先并路径压缩
        while (parent[x] != x){
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

    int Count() { return count; }
};
```

```cpp
class Difference{
    vector<int> diff;   //差分数组

public:
    /* 差分数组的初始化 */
    Difference(const vector<int>& nums){
        assert(!nums.empty());  //检查数组是否不为空
        diff.resize(nums.size());
        diff[0]=nums[0];
        for(size_t i=1;i<nums.size();i++)
            diff[i] = nums[i] - nums[i-1];
    }

    /* 差分数组的操作[val可正可负] */
    void increment(int i,int j,int val){
        assert(i >= 0 && j < (int)diff.size() && i <= j);
        diff[i]+=val;
        //确保j不是最后一个元素
        //右端点为最后一个元素时不用处理
        if(j+1<(int)diff.size())
            diff[j+1]-=val;
    }

    /* 差分数组的还原 */
    vector<int> result() const {
        vector<int> res(diff.size());
        res[0]=diff[0];
        for(size_t i=1;i<diff.size();i++)
            res[i]=res[i-1]+diff[i];
        return res;
    }
}
```

```cpp
    res.push_back(i);   // 保存当前节点
    flag[i]=true;       // 记录已访问

void bfs(int i, vector<int>& res){
    queue<int> q;       //BFS实现关键

    while(!q.empty()){
        int t=q.front();
        res.push_back(t);
        q.pop();
```

```cpp
    vector<vector<int>> map(n+1);   //邻接表
    vector<int> in(n+1);            //入度记录[关键]
    //vector<int> list;             //拓扑排序得到的顶点序列
    
    // 拓扑排序只能求得其中一种合法的顶点序列（满足边方向约束）
    // 本题使用DP来求得从源点到汇点的路径数量总和
    // 设dp[i]为从所有源点走到i的路径数量
    vector<int> dp(n+1,0);

    // 建立邻接表
    for(int i=0;i<m;i++){
        int u,v;cin>>u>>v;      //{出发点,到达点}
        map[u].push_back(v);    //加入邻接表
        in[v]++;                //记录到达点的入度
    }

    // 拓扑排序实现的关键:队列
    // 每次拿取的是入度为0的节点
    // 即队列存取的是入度为0的节点
    queue<int> zero;

    // 寻找所有入度为0的点,给寻找序列建立基础
    for(int i=1;i<=n;i++){
        if(in[i]==0){
            zero.push(i);
            // 该点入度为0
            // 即代表该点必有一条最大链
            dp[i]=1;
        }
    }

    while(!zero.empty()){
        // 先拿取一个入度为0的节点
        int u = zero.front();
        // 然后删除该节点
        zero.pop();
        // 将该节点加入序列
        //list.push_back(u);
        // 然后更新入度记录
        // 即当前由该点所连的边得到的入度全部清除
        for(int& t:map[u]){
            // 更新入度
            in[t]--;
            // 更新经由该点的路径数量
            // u节点当前为零入度点,代表必有一条路径经由
            // 故经由该点下一步可达的点的路径中必含有经由u节点的路径
            // 则下一点的路径数量必是两点路径数量之和
            dp[t] = (dp[u] + dp[t]) % MOD;
            // 如果当前点入度为0,就要将其加入队列
            if(in[t]==0) zero.push(t);
        }
    }
    
    // 所得到的list就是一种拓扑排序序列

    int ans = 0;
    // 总路径数量是所有汇点（出度为0）的路径数量之和
    // 路径末点必是出度为0的节点
    for(int i=1;i<=n;i++)
        if(map[i].empty())
            ans = (dp[i] + ans) % MOD;
    cout<<ans;
```

> Prim算法
> 

```cpp
vector<vector<pair<int,int>>> p;  //邻接表[邻居,权值]
vector<bool> flag;                //访问记录
vector<vector<int>> tree;         //最终生成的最小树

// 从一个点开始,每次选择连接该点权值最小的边
// 然后从这个边到下个点,依旧选择该点权值最小的、对应点未被访问的边
// 直到逐步构建出一个最小生成树 
int prim(int start,int n){
    vector<bool> flag(n+1,false);   //记录是否被访问过
    // 用vector作为底层容器,以pair<int,int>作为数据结构的升序最小堆队列
    priority_queue<
        tuple<int,int,int>,             //{权值,起点,终点}
        vector<tuple<int,int,int>>,     //底层容器
        greater<tuple<int,int,int>>
    > pq;
    int sum=0;              //总和
    flag[start] = true;     //标记初始点已访问
    // 存储初始点的所有边
    for(auto& edge:p[start]){
        int v=edge.first,w=edge.second;
        pq.push(make_tuple(w,start,v));
    }
    while(!pq.empty()) {
        auto t=pq.top();        // 取得连接该点中最小权值的边
        int w=get<0>(t),u=get<1>(t),v=get<2>(t);
        pq.pop();               // 去除该边
        if(flag[v]) continue;   // 如果该边对应的点已访问,就跳过
        flag[v]=true;           // 否则标记该点已访问
        sum += w;               // 计入权值
        // 加入树[无向图要加两次]
        tree[u].push_back(v);
        tree[v].push_back(u);

        // 存入对应点的所有连接的边[扩散]
        for(auto& edge:p[v]){
            int to=edge.first,w2=edge.second;
            if(!flag[to]) pq.push(make_tuple(w2,v,to));
        }
    }
    // 有效值为[1,n],注意跳过0
    if(find(flag.begin()+1,flag.end(),false)!=flag.end()) return -1;
    return sum;
}
```

```cpp
struct edge{
    int x,y,w;  //起点,终点,权值
    edge(int a,int b,int c):x(a),y(b),w(c) {}
};
vector<edge> E;             //无向图
vector<vector<int>> tree;   //最小生成树的邻接表
/* 并查集模版 */
// kruskal算法是每次选择最小的边
// 并且检查若加入该边是否会生成环
// kruskal通过并查集检查是否生成环
void kruskal(int n){
    // 排序使得每次拿取的都是最小的
    sort(E.begin(),E.end(),[](edge& a,edge& b){
        return a.w<b.w;
    });
    UnionFold uf(n);        // 并查集防止生成的树有环
    int sum = 0,cnt = 0;    // 权值总和和计数
    for(auto& e:E){
        // 合并失败就代表有连接
        if(uf.merge(e.x,e.y)){
            sum += e.w;
            tree[e.x].push_back(e.y);
            cnt++;
            if(cnt == n-1) break;
        }
    }
    // 计数不到节点数,即代表不联通
    if(cnt<n-1) cout<<"orz";
    else cout<<sum;
}
```

> Dijkstra单源最短路径【仅能在非权图中使用】
> 

```cpp
struct State {
    int node,dis;   // 当前节点,前往节点的距离[权值]
    State(int n, int d) : node(n), dis(d) {}
    // 小顶堆[确保升序第一个最小]
    bool operator<(const State& other) const {
        return dis > other.dis;
    }
};
vector<vector<pair<int,int>>> mp;
// Dijkstra:计算从原点开始每个点的带权距离
// 使用BFS的方法,即Dijkstra是BFS的加权版本
vector<int> Dijkstra(int n, int start) {
    vector<int> dis(n+1, -1);   // 从start到每个点的带权距离
    priority_queue<State> pq;   // 小顶堆[确保升序]
    pq.push(State(start, 0));
    while (!pq.empty()) {
        State t = pq.top();pq.pop();
        int v = t.node, w = t.dis;
        // 距离已更新过,就跳过
        if (dis[v] != -1) continue;
        dis[v] = w;
        for (pair<int,int>& p : mp[v]) {
            int y = p.first, w2 = w + p.second;
            if (dis[y] == -1) pq.push(State(y, w2));
        }
    }
    return dis;
}
```
