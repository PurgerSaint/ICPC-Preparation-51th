    vector<int>& nums;      //原数组

    vector<vector<int>>& mat;   //原数组
    int n=mat.size(),m=mat[0].size();
    vector<vector<int>> sum(n+1,vector<int>(m+1,0));
    /* 二维数组计算前缀和 */
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++)
            sum[i][j] = mat[i-1][j-1] + sum[i-1][j] + sum[i][j-1] - sum[i-1][j-1];
    /* 获取指定矩阵和 */
    auto getSum = [&](int r1,int c1,int r2,int c2){
        return sum[r2+1][c2+1] - sum[r2+1][c1] - sum[r1][c2+1] + sum[r1][c1];
    };

class Difference{
    vector<int> diff;   //差分数组

public:
    /* 差分数组的初始化 */
    Difference(const vector<int>& nums){
        assert(!nums.empty());  //检查数组是否不为空
        diff.resize(nums.size());
        diff[0]=nums[0];
        for(size_t i=1;i<nums.size();i++)
            diff[i] = nums[i] - nums[i-1];
    }

    /* 差分数组的操作[val可正可负] */
    void increment(int i,int j,int val){
        assert(i >= 0 && j < (int)diff.size() && i <= j);
        diff[i]+=val;
        //确保j不是最后一个元素
        //右端点为最后一个元素时不用处理
        if(j+1<(int)diff.size())
            diff[j+1]-=val;
    }

    /* 差分数组的还原 */
    vector<int> result() const {
        vector<int> res(diff.size());
        res[0]=diff[0];
        for(size_t i=1;i<diff.size();i++)
            res[i]=res[i-1]+diff[i];
        return res;
    }
}

# 力扣303.区域和检索 - 数组不可变

给定一个整数数组nums，要求查询[left,right]区间的和。

首先使用了`sum[right]-sum[left-1]`，但此时会遇到`left=0`的情况，故只能对前缀和数组后移，求和也相对后移。

```cpp
class NumArray {
public:
    vector<int> sum;
    NumArray(vector<int>& nums) {
        if(!nums.empty()){
            int n=nums.size();
            sum.resize(n+1,0);
            for(int i=0;i<n;i++)
                sum[i+1]=sum[i]+nums[i];
        }
    }
    
    int sumRange(int left, int right) {
        return sum[right+1]-sum[left];
    }
};
```

# 力扣304.**二维区域和检索 - 矩阵不可变**

给定一个二维矩阵matrix，求左上角[row1,col1]、左下角[row2,col2]的矩阵和。

同样采用坐标偏移，计算前缀和时公式为：

$当前位置的值+左边矩阵和+上边矩阵和-左边矩阵和上边矩阵重复矩阵的和$

计算固定矩阵的和公式为：

$右下角到原点的整个矩形 - 左边的矩形 - 上边的矩形 + 左上角被重复减掉的矩形$

```cpp
class NumMatrix {
public:
    vector<vector<int>> sum;
    NumMatrix(vector<vector<int>>& matrix) {
        int n=matrix.size(),m=matrix[0].size();
        if(n==0||m==0) return;
        sum.resize(n+1,vector<int>(m+1,0));
        /* 上矩形+左矩形-重复矩形 */
        for(int i=1;i<=n;i++)
            for(int j=1;j<=m;j++)
                sum[i][j]=matrix[i-1][j-1]+sum[i][j-1]+sum[i-1][j]-sum[i-1][j-1];
    }
    
    int sumRegion(int r1, int c1, int r2, int c2) {
        /* 采用了坐标偏移 */
        return sum[r2+1][c2+1]     //整个矩形
              -sum[r2+1][c1]       //去掉左边的矩形
              -sum[r1][c2+1]       //去掉上面的矩形
              +sum[r1][c1];        //再加回左上角[重复减了]
    }
};
```

# 力扣1314.矩阵区域和

给你一个 `m x n` 的矩阵 `mat` 和一个整数 `k` ，请你返回一个矩阵 `answer` ，其中每个 `answer[i][j]` 是所有满足下述条件的元素 `mat[r][c]` 的和：

- `i - k <= r <= i + k,`
- `j - k <= c <= j + k` 且
- `(r, c)` 在矩阵内。

**示例 1：**

```
输入：mat = [[1,2,3],[4,5,6],[7,8,9]], k = 1
输出：[[12,21,16],[27,45,33],[24,39,28]]

```

**示例 2：**

```
输入：mat = [[1,2,3],[4,5,6],[7,8,9]], k = 2
输出：[[45,45,45],[45,45,45],[45,45,45]]
```

二维前缀和的应用。计算前缀和，构造对应矩阵并返回即可。

```cpp
class Solution {
public:
    vector<vector<int>> matrixBlockSum(vector<vector<int>>& mat, int k) {
        int n=mat.size(),m=mat[0].size();
        vector<vector<int>> sum(n+1,vector<int>(m+1,0));
        /* [模版]二维数组计算前缀和 */
        for(int i=1;i<=n;i++)
            for(int j=1;j<=m;j++)
                sum[i][j] = mat[i-1][j-1] + sum[i-1][j] + sum[i][j-1] - sum[i-1][j-1];
        /* [模版]获取指定矩阵和 */
        auto getSum = [&](int r1,int c1,int r2,int c2){
            return sum[r2+1][c2+1] - sum[r2+1][c1] - sum[r1][c2+1] + sum[r1][c1];
        };
        vector<vector<int>> res(n,vector<int>(m,0));
        for(int i=0;i<n;i++)
            for(int j=0;j<m;j++){
                int x1 = max(i-k,0);
                int y1 = max(j-k,0);

                int x2 = min(i+k,n-1);
                int y2 = min(j+k,m-1);

                res[i][j]=getSum(x1,y1,x2,y2);
            }
        return res;
    }
};
```

# **力扣724. 寻找数组的中心下标**

给定一个数组nums，获取其最左侧的中心下标，中心下标代表其左侧所有元素相加的和等于其右侧所有元素相加的和。

一维前缀和的应用。构建前缀和，然后从左侧开始判断即可。

```cpp
class Solution {
public:
    int pivotIndex(vector<int>& nums) {
        int n=nums.size();
        vector<int> sum(n+1,0);
        for(int i=0;i<n;i++)
            sum[i+1]=sum[i]+nums[i];
        auto getSum = [&](int left,int right){
            return sum[right+1]-sum[left];
        };
        int res=-1;
        for(int i=0;i<n;i++)
            if(getSum(0,i-1)==getSum(i+1,n-1)){
                res=i;
                break;
            }
        return res;
    }
};
```

# 力扣238.除自身以外数组的乘积

给定一个数组nums，求由除当前元素外剩余元素之积组成的数组。

一维前缀和的扩展。题目要求不能使用除法，就通过前缀积乘以后缀积即可得到结果。

```cpp
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n=nums.size();
        //从左到右的前缀积
        vector<int> prefix(n);
        prefix[0]=nums[0];
        for(int i=1;i<n;i++)
            prefix[i]=prefix[i-1]*nums[i];

        //从右到左的后缀积
        vector<int> suffix(n);
        suffix[n-1]=nums[n-1];
        for(int i=n-2;i>=0;i--)
            suffix[i]=suffix[i+1]*nums[i];

        vector<int> res(n);
        res[0]=suffix[1];
        res[n-1]=prefix[n-2];
        //除当前i的积就是前缀积*后缀积
        for(int i=1;i<n-1;i++)
            res[i]=prefix[i-1]*suffix[i+1];
        return res;
    }
};
```

# 力扣**1352. 最后 K 个数的乘积**

设计一个算法，该算法接受一个整数流并检索该流中最后 `k` 个整数的乘积。

实现 `ProductOfNumbers` 类：

- `ProductOfNumbers()` 用一个空的流初始化对象。
- `void add(int num)` 将数字 `num` 添加到当前数字列表的最后面。
- `int getProduct(int k)` 返回当前数字列表中，最后 `k` 个数字的乘积。你可以假设当前列表中始终 **至少** 包含 `k` 个数字。

题目数据保证：任何时候，任一连续数字序列的乘积都在 32 位整数范围内，不会溢出。

前缀积的扩展。$[left,right](0<left≤right≤n-1)$范围内所有元素的积等于$fix[right]/fix[left-1]$，扩展即可。

```cpp
class ProductOfNumbers {
public:
    vector<int> suffix;

    ProductOfNumbers() {
        suffix.push_back(1);
    }
    
    void add(int num) {
        if(num==0){
            suffix.clear();
            suffix.push_back(1);
            return;
        }
        int n=suffix.size();
        suffix.push_back(suffix[n-1]*num);
    }
    
    int getProduct(int k) {
        int n=suffix.size();
        if(k>n-1) return 0;
        return suffix[n-1]/suffix[n-1-k];
    }
};
```

# 力扣560. 和为 K 的子数组

给你一个整数数组 `nums` 和一个整数 `k` ，请你统计并返回该数组中和为 `k` ****的子数组的个数 ，子数组是数组中元素的连续非空序列。

一维前缀和加上哈希表的使用。原版使用双循环会超时，加上了哈希表进行优化。

```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int n=nums.size();
        vector<int> sum(n+1,0); //前缀和数组
        unordered_map<int,int> hash;    //值:出现次数
        hash[0]=1;
        int res=0;
        for(int i=1;i<=n;i++){
            sum[i]=sum[i-1]+nums[i-1];
            int need=sum[i]-k;
            if(hash.find(need)!=hash.end())
                res+=hash[need];
            hash[sum[i]]++;
        }
        return res;
    }
};
```

# 力扣1109. 航班预订统计

这里有 `n` 个航班，它们分别从 `1` 到 `n` 进行编号。

有一份航班预订表 `bookings` ，表中第 `i` 条预订记录 `bookings[i] = [firsti, lasti, seatsi]` 意味着在从 `firsti` 到 `lasti` （**包含** `firsti` 和 `lasti` ）的 **每个航班** 上预订了 `seatsi` 个座位。

请你返回一个长度为 `n` 的数组 `answer`，里面的元素是每个航班预定的座位总数。

题目大意就是，告诉你n个航班中，分别在i到j的航班定了座位，统计每个航班的已售出座位。就是直接使用差分，统计区间，然后还原即可。

```cpp
class Solution {
public:
    class Difference{
        vector<int> diff;   //差分数组

    public:
        Difference(const vector<int>& nums){
            assert(!nums.empty());  //检查数组是否不为空
            diff.resize(nums.size());
            diff[0]=nums[0];
            for(size_t i=1;i<nums.size();i++)
                diff[i] = nums[i] - nums[i-1];
        }

        void increment(int i,int j,int val){
            diff[i]+=val;
            //确保j不是最后一个元素
            //最后一个元素不用处理左端点
            if(j+1<diff.size())
                diff[j+1]-=val;
        }

        vector<int> result(){
            vector<int> res(diff.size());
            res[0]=diff[0];
            for(size_t i=1;i<diff.size();i++)
                res[i]=res[i-1]+diff[i];
            return res;
        }
    };

    vector<int> corpFlightBookings(vector<vector<int>>& bookings, int n) {
        vector<int> nums(n,0);
        Difference df(nums);

        for(const auto& booking:bookings){
            int i = booking[0] - 1;
            int j = booking[1] - 1;
            int val = booking[2];
            df.increment(i,j,val);
        }
        return df.result();
    }
};
```

# 力扣1094. 拼车

车上最初有 `capacity` 个空座位。车 **只能** 向一个方向行驶（也就是说，**不允许掉头或改变方向**）

给定整数 `capacity` 和一个数组 `trips` ,  `trip[i] = [numPassengersi, fromi, toi]` 表示第 `i` 次旅行有 `numPassengersi` 乘客，接他们和放他们的位置分别是 `fromi` 和 `toi` 。这些位置是从汽车的初始位置向东的公里数。

当且仅当你可以在所有给定的行程中接送所有乘客时，返回 `true`，否则请返回 `false`。

差分数组专用题。乘客会在`fromi`站上车，`toi`站下车，即`[fromi, toi]`站有乘客`numPassengersi`，使用差分进行区间操作，最后统计是否有超出座位的乘客即可。

```cpp
class Solution {
public:
    class Difference{
        vector<int> diff;   //差分数组

    public:
        Difference(const vector<int>& nums){
            assert(!nums.empty());  //检查数组是否不为空
            diff.resize(nums.size());
            diff[0]=nums[0];
            for(size_t i=1;i<nums.size();i++)
                diff[i] = nums[i] - nums[i-1];
        }

        void increment(int i,int j,int val){
            diff[i]+=val;
            //确保j不是最后一个元素
            //最后一个元素不用处理左端点
            if(j+1<diff.size())
                diff[j+1]-=val;
        }

        vector<int> result(){
            vector<int> res(diff.size());
            res[0]=diff[0];
            for(size_t i=1;i<diff.size();i++)
                res[i]=res[i-1]+diff[i];
            return res;
        }
    };

    bool carPooling(vector<vector<int>>& trips, int capacity) {
        vector<int> nums(1001,0);
        Difference df(nums);

        for(const auto& trip:trips){
            int i = trip[1];
            int j = trip[2]-1;
            int val = trip[0];
            df.increment(i,j,val);
        }

        vector<int> res=df.result();
        for(int& i:res) if(i>capacity) return false;
        return true;
    }
};
```
