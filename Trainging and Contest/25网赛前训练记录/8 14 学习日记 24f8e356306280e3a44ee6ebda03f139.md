```cpp
/* 并查集完全版 */
class UnionFold {
private:
    /* 连通分量[不同集合]个数 */
    int count;              //连通分量个数
    /* 每个节点的上一节点[祖先] */
    vector<int> parent;     //祖先
    /* 以每个节点为根节点的树大小 */
    vector<int> size;       //大小

public:
    // 初始化并查集
    UnionFold(int n) : count(n), parent(n), size(n, 1) {
        for (int i = 0; i< n; i++) {
            parent[i]=i;
        }
    }

    // 将 p 和 q 连通
    void unionSets(int p, int q){
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP == rootQ) return;

        // 以每个节点为根节点的树大小作为权重
        // 将小树连接到大树下面
        if (size[rootP] > size[rootQ]) {
            parent[rootQ] = rootP;
            size[rootP] += size[rootQ];
        } else {
            parent[rootP] = rootQ;
            size[rootQ] += size[rootP];
        }
        count--;
    }

    // 判断 p 和 q 是否互相连通[在同一集合内]
    bool connected(int p, int q) {
        return find(q) == find(p);
    }

    // 寻找 x 所在树的根节点
    int find(int x) {
        //寻找x的祖先
        while (parent[x] != x){
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }

    int Count() { return count; }
};
/* 简略版并查集 */
class UnionFold_Ver2 {
private:
    vector<int> bcj;

public:
    int find(int a) {
        return bcj[a]==a?a:(bcj[a]=find(bcj[a]));
    }
    
    void merge(int a, int b) {
        int fx = find(a), fy = find(b);
        if (fx != fy) bcj[fx] = fy;
    }
}
```

力扣26. 删除有序数组中的重复项

给定一个递增数组`nums`，要求在`nums`中原地删除重复元素，返回删除后的数组长度。

可以使用快慢指针，快指针用于寻找数组中第一个没有重复的，当找到没有重复的，就让快慢指针所指引的数进行交换，使得所有重复的数都会在数组末尾，类似手动实现unique方法。

```cpp
class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int n=nums.size();
        if(n==0) return 0;
        int fast=0,slow=0;
        while(fast<n){
          if(nums[fast]!=nums[slow]){
            slow++;
            nums[slow]=nums[fast];
          }
          fast++;
        }
        return slow+1;
    }
};
```

力扣27.移除元素

给定一个递增数组`nums`，要求在`nums`中原地删除值为val的元素，返回数组中值不等于val的元素个数。

同样的快慢指针，将所有val移到后面即可。

```cpp
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int n=nums.size();
        if(n==0) return 0;
        int slow=0,fast=0;
        while(fast<n){
            if(nums[fast]!=val){
                nums[slow]=nums[fast];
                slow++;
            }
            fast++;
        }
        return slow;
    }
};
```

力扣283.移动零

给定一个数组 `nums`，编写一个函数将所有 `0` 移动到数组的末尾，同时保持非零元素的相对顺序。

同样的快慢指针，将所有非0元素移动到数组前面即可，但注意，移动是复制，没有改变原有元素的值，需要对后面移动过的元素赋0。

```cpp
class Solution {
public:
    void moveZeroes(vector<int>& nums) {
        int fast=0,slow=0;
        while(fast<nums.size()){
            if(nums[fast]!=0){
                nums[slow]=nums[fast];
                slow++;
            }
            fast++;
        }
        for(int i=slow;i<nums.size();i++)
            nums[i]=0;
    }
};
```

力扣**167. 两数之和 II - 输入有序数组**

给与一个**非递减顺序排列**的数组，请返回数组中两个元素之和为target的下标（下标从1开始）。

左右指针从头尾开始寻找即可，若相加之和比target大，则减小右指针；若小，则增大左指针。

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int left=0,right=numbers.size()-1;
        while(left<right){
            int sum=numbers[left]+numbers[right];
            if(sum==target)
                return vector<int>{left+1,right+1};
            else if(sum<target) left++;
            else right--;
        }
        return vector<int>{-1,-1};
    }
};
```

力扣76.最小覆盖子串

给你一个字符串 `s` 、一个字符串 `t` 。返回 `s` 中涵盖 `t` 所有字符的最小子串。如果 `s` 中不存在涵盖 `t` 所有字符的子串，则返回空字符串 `""` 。

使用双指针的滑动窗口方法实现。维护每一个含有字符串t内字符的滑动窗口字符串，然后通过缩短该滑动窗口字符串（右移左指针），寻找是否存在一个最小字符串。

```cpp
class Solution {
    public:
    string minWindow(string s, string t) {
        unordered_map<char,int> need,window;
        for(char& c:t) need[c]++;

        int left=0,right=0;
        //窗口内有效字符个数
        int vaild=0;
        //最小子串的起始位置和长度
        int start=0,len=INT_MAX;
        while(right<s.size()){
            //当前要检查的字符
            char& c=s[right];
            //右移右指针
            right++;
            //如果该字符存在目标字符串内
            if(need.count(c)){
                //更新窗口内字符数据
                window[c]++;
                //并更新有效字符个数
                if(window[c]==need[c])
                    vaild++;
            }

            //寻找最小覆盖子串
            while(vaild==need.size()){
                //更新最小覆盖子串的长度及位置
                if(right-left<len){
                    start=left;
                    len=right-left;
                }
                //要移出的字符
                char d=s[left];
                //右移左指针
                left++;
                //更新窗口内数据
                if(need.count(d)){
                    if(window[d]==need[d])
                        vaild--;
                    window[d]--;
                }
            }
        }
        return len==INT_MAX?"":s.substr(start,len);
    }
};
```

力扣**567. 字符串的排列**

给你两个字符串 `s1` 和 `s2` ，写一个函数来判断 `s2` 是否包含 `s1` ****的 排列。如果是，返回 `true` ；否则，返回 `false` 。

同样使用滑动窗口方法。上题改变滑动窗口左指针的条件是找到了一个含有目标字符串内所有字符的子字符串，故而要通过缩小，来判断是否存在一个更小的、符合条件的子字符串；而本题的条件是当滑动窗口的大小超过目标字符串时，就直接进行改变，因为题目要求的是寻找是否存在，而并非更小。

```cpp
class Solution {
public:
    bool checkInclusion(string s1, string s2) {
        unordered_map<char,int> window,need;
        for(char& c:s1) need[c]++;

        int left=0,right=0;
        int vaild=0;
        while(right<s2.size()){
          char c=s2[right];
          right++;
          if(need.count(c)){
            window[c]++;
            if(window[c]==need[c])
              vaild++;
          }

          while(right - left >= s1.size()){
            if (vaild == need.size())
              return true;
            char f=s2[left];
            left++;
            if(need.count(f)){
              if(window[f]==need[f])
                vaild--;
              window[f]--;
            }
          }
        }
        return false;
    }
};
```

力扣3.无重复字符的最长子串

给定一个字符串 `s` ，请你找出其中不含有重复字符的 **最长 子串** 的长度。

同样的滑动窗口方法，改变左指针的条件是当前计入的字符有重复的，当更新完左右指针后就可以更新值了。

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_map<char, int> window;

        int left = 0, right = 0;
        int res = 0;
        while (right < s.size()) {
            char c = s[right];
            right++;
            window[c]++;

            while (window[c] > 1) {
                char d=s[left];
                left++;
                window[d]--;
            }
            res = max(res, right-left);
        }
        return res;
    }
};
```

力扣**990. 等式方程的可满足性**

给定一个由表示变量之间关系的字符串方程组成的数组，每个字符串方程 `equations[i]` 的长度为 `4`，并采用两种不同的形式之一：`"a==b"` 或 `"a!=b"`。在这里，a 和 b 是小写字母（不一定不同），表示单字母变量名。

只有当可以将整数分配给变量名，以便满足所有给定的方程时才返回 `true`，否则返回 `false`。

看似复杂，其实使用并查集解决即可。当两个变量相等时，进行合并；两个变量不等时，使用并查集判断他们是否在同一个集合之内，不在直接返回false即可。

```cpp
class Solution {
public:
    class UnionFold {
    private:
        int count;
        vector<int> parent;
        vector<int> size;

    public:
        UnionFold(int n):count(n), parent(n), size(n, 1) {
            for (int i = 0; i < n; i++)
                parent[i] = i;
        }

        bool connected(int p, int q) {
            return find(p) == find(q);
        }

        void merge(int p, int q) {
            int rootP = find(p);
            int rootQ = find(q);
            if (rootP == rootQ) return;

            if (size[rootP] > size[rootQ]) {
                parent[rootQ] = rootP;
                size[rootP] += size[rootQ];
            }else {
                parent[rootP] = rootQ;
                size[rootQ] += size[rootP];
            }
            count--;
        }

        int find(int x) {
            while (parent[x] != x) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }
    };
    bool equationsPossible(vector<string>& equations) {
        UnionFold uf(26);
        for (string& s:equations) {
            char x = s[0];
            char y = s[3];
            if (s[1] == '=') 
                uf.merge(x - 'a', y - 'a');
        }
        for (string& s:equations) {
            char x = s[0];
            char y = s[3];
            if (s[1] == '!'&&uf.connected(x - 'a', y - 'a'))
                return false;
        }
        return true;
    }
};
```

力扣**547. 省份数量**

有 `n` 个城市，其中一些彼此相连，另一些没有相连。如果城市 `a` 与城市 `b` 直接相连，且城市 `b` 与城市 `c` 直接相连，那么城市 `a` 与城市 `c` 间接相连。

**省份** 是一组直接或间接相连的城市，组内不含其他没有相连的城市。

给你一个 `n x n` 的矩阵 `isConnected` ，其中 `isConnected[i][j] = 1` 表示第 `i` 个城市和第 `j` 个城市直接相连，而 `isConnected[i][j] = 0` 表示二者不直接相连。

返回矩阵中 **省份** 的数量。

类似于不同门派，直接套用并查集模版即可。

```cpp
class Solution {
public:
    class UnionFold {
        private:
            int count;
            vector<int> parent;
            vector<int> size;
        public:
            UnionFold(int n): count(n), parent(n), size(n) {
                for(int i = 0; i < n; i++) {
                    parent[i] = i;
                }
            }

            void merge(int p, int q) {
                int rootP = find(p);
                int rootQ = find(q);
                if (rootQ == rootP) return;

                if (size[rootP] > size[rootQ]) {
                    parent[rootQ] = parent[rootP];
                    size[rootP] += size[rootQ];
                } else {
                    parent[rootP] = parent[rootQ];
                    size[rootQ] += size[rootP];
                }
                count--;
            }

            int find(int x) {
                while (x != parent[x]) {
                    parent[x] = parent[parent[x]];
                    x = parent[x];
                }
                return x;
            }

            int Count(){ return count; }
    };
    int findCircleNum(vector<vector<int>>& isConnected) {
        int n=isConnected.size();
        UnionFold uf(n);
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (isConnected[i][j] == 1)
                    uf.merge(i, j);
        return uf.Count();
    }
};
```
