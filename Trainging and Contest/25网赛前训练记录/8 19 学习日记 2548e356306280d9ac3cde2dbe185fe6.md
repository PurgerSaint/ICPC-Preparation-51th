洛谷**P4017 最大食物链计数**

给你一个食物网，你要求出这个食物网中最大食物链的数量。

（这里的“最大食物链”，指的是**生物学意义上的食物链**，即**最左端是不会捕食其他生物的生产者，最右端是不会被其他生物捕食的消费者**。）

Delia 非常急，所以你只有 1 秒的时间。

由于这个结果可能过大，你只需要输出总数模上 80112002 的结果。

> 拓扑排序（Kanh入度法）
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
const int MOD = 80112002;
int main(){
    IOS;
    int n,m;cin>>n>>m;
    /* 邻接表 */
    vector<vector<int>> map(n+1);
    /* 入度记录[关键] */
    vector<int> in(n+1);
    /* 拓扑排序得到的顶点序列 */
    //vector<int> list;
    
    // 拓扑排序只能求得其中一种合法的顶点序列（满足边方向约束）
    // 本题使用DP来求得从源点到汇点的路径数量总和
    // 设dp[i]为从所有源点走到i的路径数量
    vector<int> dp(n+1,0);

    // 建立邻接表
    for(int i=0;i<m;i++){
        int u,v;    //{出发点,到达点}
        cin>>u>>v;
        // 加入邻接表
        map[u].push_back(v);
        // 记录到达点的入度
        in[v]++;
    }

    // 拓扑排序实现的关键:队列
    // 每次拿取的是入度为0的节点
    // 即队列存取的是入度为0的节点
    queue<int> zero;

    // 寻找所有入度为0的点,给寻找序列建立基础
    for(int i=1;i<=n;i++){
        if(in[i]==0){
            zero.push(i);
            // 该点入度为0
            // 即代表该点必有一条最大链
            dp[i]=1;
        }
    }

    while(!zero.empty()){
        // 先拿取一个入度为0的节点
        int u = zero.front();
        // 然后删除该节点
        zero.pop();
        // 将该节点加入序列
        //list.push_back(u);
        // 然后更新入度记录
        // 即当前由该点所连的边得到的入度全部清除
        for(int& t:map[u]){
            // 更新入度
            in[t]--;
            // 更新经由该点的路径数量
            // u节点当前为零入度点,代表必有一条路径经由
            // 故经由该点下一步可达的点的路径中必含有经由u节点的路径
            // 则下一点的路径数量必是两点路径数量之和
            dp[t] = (dp[u] + dp[t]) % MOD;
            // 如果当前点入度为0,就要将其加入队列
            if(in[t]==0) zero.push(t);
        }
    }
    
    // 所得到的list就是一种拓扑排序序列

    int ans = 0;
    // 总路径数量是所有汇点（出度为0）的路径数量之和
    // 路径末点必是出度为0的节点
    for(int i=1;i<=n;i++)
        if(map[i].empty())
            ans = (dp[i] + ans) % MOD;
    cout<<ans;

    return 0;
}
```

> 拓扑排序（DFS回溯法）
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
const int MOD = 80112002;
/* 邻接表 */
vector<vector<int>> mp;
/* 拓扑排序得到的顶点序列 */
vector<int> lt;
/* 每个点的访问记录 */
vector<bool> flag;
vector<int> in;
vector<int> dp;
void dfs(int u){
    flag[u]=true;
    for(int& v:mp[u])
        if(!flag[v]){
            dfs(v);
        }
    // 回溯时加入序列
    lt.push_back(u);
}
int main(){
    IOS;
    int n,m;cin>>n>>m;
    mp = vector<vector<int>>(n+1);
    flag = vector<bool>(n+1,false);
    dp = vector<int>(n+1,0);
    in = vector<int>(n+1,0);

    for(int i=0;i<m;i++){
        int u,v;cin>>u>>v;
        mp[u].push_back(v);
        in[v]++;
    }

    // 找到源点,并将其初始化为1
    for(int i=1;i<=n;i++)
        if(in[i]==0) dp[i]=1;
    
    for(int i=1;i<=n;i++)
        if(!flag[i]) dfs(i);
    
    // 回溯法得到的序列是倒着的拓扑序
    reverse(lt.begin(),lt.end());

    for(int u:lt){
        for(int v:mp[u]){
            dp[v] = (dp[u] + dp[v]) % MOD;
        }
    }

    int ans = 0;
    for(int i=1;i<=n;i++)
        if(mp[i].empty())
            ans = (ans + dp[i]) % MOD;
    cout<<ans;

    return 0;
}
```

力扣**36. 二叉树的最近公共祖先**

给定一个二叉树, 找到该树中两个指定节点的最近公共祖先。

> DFS递增法
> 

```cpp
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if(root == NULL) return NULL;
        if(root->val == p->val || root->val == q->val) return root;
        TreeNode* left = lowestCommonAncestor(root->left,p,q);
        TreeNode* right = lowestCommonAncestor(root->right,p,q);
        if(left != NULL && right != NULL){
            return root;
        }
        return left!=NULL?left:right;
    }
};
```

力扣46.全排列

给定一个不含重复数字的数组 `nums` ，返回其 *所有可能的全排列* 。你可以 **按任意顺序** 返回答案。

选择每一个数字组成一个排列的过程中，就像是构建一颗决策树。而访问这个决策树的子树返回时，就是回溯的真正实现。

```cpp
class Solution {
private:
    vector<bool> flag;          // 记录是否被选择过
    vector<vector<int>> res;    // 结果
public:
    void backtrack(vector<int>& nums,deque<int>& path){
        // 递归结束条件
        if(path.size() == nums.size()){
            res.push_back(vector<int>(path.begin(),path.end()));
            return;
        }

        // 做决策树:是否把下一个数加入到选择中
        for(int i=0;i<nums.size();i++){
            // 被选择过,跳过
            if(flag[i]) continue;
            // 将当前数加入到选择中
            path.push_back(nums[i]);
            flag[i] = true;
            // 进入下一层决策树
            backtrack(nums,path);
            // 撤销当前选择
            path.pop_back();
            flag[i] = false;
        }  
    }
    vector<vector<int>> permute(vector<int>& nums) {
        deque<int> path;
        flag = vector<bool>(nums.size()+1,false);
        backtrack(nums,path);
        return res;
    }
};
```

力扣37.解数独

编写一个程序，通过填充空格来解决数独问题。

数独的解法需 **遵循如下规则**：

1. 数字 `1-9` 在每一行只能出现一次。
2. 数字 `1-9` 在每一列只能出现一次。
3. 数字 `1-9` 在每一个以粗实线分隔的 `3x3` 宫内只能出现一次。（请参考示例图）

数独部分空格内已填入了数字，空白格用 `'.'` 表示。

```cpp
class Solution {
    private:
        // 标记是否找到可行解
        bool found;
        // 记录每行出现的数字
        vector<unordered_set<char>> rows;
        // 记录每列出现的数字
        vector<unordered_set<char>> cols;
        // 记录每个九宫格已经出现的数字
        vector<unordered_set<char>> boxes; 
    public:
        Solution(): found(false), rows(9), cols(9), boxes(9) {}

        void solveSudoku(vector<vector<char>>& board) {
            /*
                填数独就像是在枚举一个长宽均为9的二维数组中每一个的可能选择
                类似于做一个决策树,这个决策树中每一个节点都有9个孩子,寻找这个决策树中正确的一条路径
                决策树中有的子树是没必要存在的,就可以通过剪枝进行优化
                回溯是为了防止错过正确路径
            */
            // 初始记录
            for (int i = 0; i < 9; i++) {
                for (int j = 0; j < 9; j++) {
                    if (board[i][j] != '.') {
                        rows[i].insert(board[i][j]);
                        cols[j].insert(board[i][j]);
                        boxes[getBoxIndex(i, j)].insert(board[i][j]);
                    }
                }
            }
            backtrack(board, 0);
        }
    private:
        // {行,列}坐标转换为九宫格中的小宫格序号
        int getBoxIndex(int r, int c){
            return (r / 3) * 3 + (c / 3);
        }
        void backtrack(vector<vector<char>>& board, int index) {
            // 找到可行解,就立即结束
            if (found) return;
            // 行列大小
            int n = 9, m = 9;
            // 当前操作的行列坐标
            int i = index / n, j = index % n;
            // 当递归到边界(n*m),代表所有格子都被填过
            // 并且一定找到了一组解
            if (index == m*n) {
                found = true;
                return;
            }
            // 如果有预设数字,就跳过本层进入下一层
            if (board[i][j] != '.') {
                backtrack(board, index + 1);
                return;
            }
            // 进行决策树选择,枚举当前节点中每个孩子是否可行
            for (char c = '1'; c <= '9'; c++) {
                // 如果当前数不适合放在当前格子中,就跳过
                // 避免决策树递归过长[剪枝]
                if (!isValid(i, j, c)) continue;
                // 做选择,选择当前该填入的数
                board[i][j] = c;
                rows[i].insert(c);
                cols[j].insert(c);
                boxes[getBoxIndex(i, j)].insert(c);
                // 进入下一层决策树
                backtrack(board, index + 1);
                // 如果可行,不能撤销
                if (found) return;
                // 撤销选择[回溯]
                board[i][j] = '.';
                rows[i].erase(c);
                cols[j].erase(c);
                boxes[getBoxIndex(i, j)].erase(c);
            }
        }
        // 判断是否可以在(r,c)位置放入数组num
        bool isValid(int r, int c, char num) {
            if (rows[r].count(num)) return false;
            if (cols[c].count(num)) return false;
            if (boxes[getBoxIndex(r, c)].count(num)) return false;
            return true;
        }
    };
```

力扣51.N皇后

按照国际象棋的规则，皇后可以攻击与之处在同一行或同一列或同一斜线上的棋子。

**n 皇后问题** 研究的是如何将 `n` 个皇后放置在 `n×n` 的棋盘上，并且使皇后彼此之间不能相互攻击。

给你一个整数 `n` ，返回所有不同的 **n **皇后问题** 的解决方案。

每一种解法包含一个不同的 **n 皇后问题** 的棋子放置方案，该方案中 `'Q'` 和 `'.'` 分别代表了皇后和空位。

```cpp
class Solution {
private:
    vector<vector<string>> res;
public:
    vector<vector<string>> solveNQueens(int n) {
        // 初始化空棋盘
        vector<string> board(n, string(n, '.'));
        backtrack(board, 0);
        return res;
    }
private:
    /*
        N皇后初想法是建立n*n的二维数组,然后枚举每个[i][j]的皇后可能性.
        但时间太长,决策树叶子太多.
        所以要进行第一步剪枝,只用考虑每一行的皇后可能性,不必考虑整行整列.
        然后在回溯中枚举该行每一位置的可能性.
        如果当前位置不能放皇后,即同行有皇后冲突,则跳过,即第二步剪枝优化.
    */
    void backtrack(vector<string>& board, int row) {
        // DFS递归终点:大小达到
        if(row == board.size()) {
            res.push_back(board);
            return;
        }
        int n = board[row].size();
        for (int col = 0; col < n; col++) {
            // 剪枝优化:排除不合法的选择
            if (!isValid(board, row, col)) continue;
            // 做选择
            board[row][col] = 'Q';
            // 进入下一层
            backtrack(board, row + 1);
            // 回溯:撤销选择
            board[row][col] = '.';
        }
    }
    // 是否可以在 board[row][col] 放置皇后
    bool isValid(const vector<string>& board, int row, int col) {
        int n = board.size();
        // 检查列是否有冲突
        for (int i = 0; i < n; i++) {
            if (board[i][col] == 'Q')
                return false;
        }
        // 检查右上方是否有皇后互相冲突
        for (int i = row - 1, j = col + 1;i >= 0 && j < n; i--, j++) {
            if(board[i][j] == 'Q')
                return false;
        }
        // 检查左上方是否有皇后互相冲突
        for (int i = row - 1, j = col - 1;i >= 0 && j >= 0; i--, j--) {
            if(board[i][j] == 'Q')
                return false;
        }
        return true;
    }
};
```

力扣52.N 皇后 II

**n 皇后问题** 研究的是如何将 `n` 个皇后放置在 `n × n` 的棋盘上，并且使皇后彼此之间不能相互攻击。

给你一个整数 `n` ，返回 **n 皇后问题** 不同的解决方案的数量。

```cpp
class Solution {
public:
    int totalNQueens(int n) {
        vector<string> board(n, string(n, '.'));
        return backtrack(board, 0);
    }
private:
    int backtrack(vector<string>& board, int row) {
        if (board.size() == row) return 1;
        int n = board[row].size();
        int res = 0;
        for (int col = 0; col < n; col++) {
            if (!isValid(board, row, col)) continue;
            board[row][col] = 'Q';
            res += backtrack(board, row + 1);
            board[row][col] = '.';
        }
        return res;
    }
    bool isValid(const vector<string>& board, int row, int col) {
        int n = board.size();
        for (int i = 0; i < n; i++)
            if (board[i][col] == 'Q')
                return false;
        for (int i = row - 1, j = col - 1; i >= 0 && j >= 0; i--, j--)
            if (board[i][j] == 'Q')
                return false;
        for (int i = row - 1, j = col + 1; i >= 0 && j < n; i--, j++)
            if (board[i][j] == 'Q')
                return false;
        return true;
    }
};
```
