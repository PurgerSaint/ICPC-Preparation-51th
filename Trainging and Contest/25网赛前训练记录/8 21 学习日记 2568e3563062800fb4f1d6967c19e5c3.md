```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
int n,m;
vector<vector<pair<int,int>>> mp;
vector<bool> flag;
int prim(int start) {
    int sum = 0;
    // 错误1:忘记Prim的数据结构是升序的最大堆
    priority_queue<
        tuple<int,int,int>,
        vector<tuple<int,int,int>>,
        greater<tuple<int,int,int>>
    > q;
    
    flag[start] = true;
    for (auto& edge:mp[start]) {
        int v = edge.first, w = edge.second;
        q.push(make_tuple(w, start, v));
    }

    while(!q.empty()) {
        auto t = q.top();
        int u = get<1>(t), v = get<2>(t), w = get<0>(t);
        q.pop();
        if (flag[v]) continue;
        flag[v] = true;
        sum += w;
        for (auto& edge : mp[v]) {
            int y = edge.first, z = edge.second;
            if (!flag[y]) q.push(make_tuple(z, v, y));
        }
    }

    // 错误2:忘记flag[0]永远为false
    if (find(flag.begin() + 1, flag.end(), false) != flag.end()) return -1;
    return sum;
}

    IOS;
    cin>>n>>m;
    mp = vector<vector<pair<int,int>>>(n+1);
    flag = vector<bool>(n+1, false);
    for (int i = 0; i < m; i++) {
        int x, y, z;cin>>x>>y>>z;
        mp[x].push_back({y, z});
        mp[y].push_back({x, z});
        // 错误3:将其与拓扑排序搞混
    }
```

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
struct edge {
    int u,v,w;
    edge(int a, int b, int c):u(a), v(b), w(c) {}
    // 多余1:使用vector比最大堆高效,最大堆需要重载运算符
    bool operator<(const edge& p) const {
        return w > p.w;
    }
};
priority_queue<edge> q;
class UnionFold {
    private:
        vector<int> parent;
        vector<int> size;
    public:
        UnionFold(int n):parent(n+1),size(n+1, 1) {
            for (int i = 0; i <= n; i++)
                parent[i] = i;
        }

        bool merge(int p, int q) {
            int rootP = find(p);
            int rootQ = find(q);
            if (rootP == rootQ) return false;
        
            if (size[rootQ] > size[rootP]) {
                size[rootQ] += size[rootP];
                parent[rootP] = rootQ;
            } else {
                size[rootP] += size[rootQ];
                parent[rootQ] = rootP;
            }
            return true;
        }

        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }
};
int Kruskal(int n) {
    int sum = 0, cnt = 0;
    UnionFold uf(n);
    while (!q.empty()) {
        edge t=q.top();
        int x = t.u, y = t.v, w = t.w;
        q.pop();
        // 错误1:忘记Kruskal用的是并查集进行判断成环
        if (uf.merge(x, y)) {
            sum += w;
            cnt++;
            if (cnt == n-1) break;
        }
    }
    return cnt < n-1 ? -1 : sum;
}
int main() {
    IOS;
    int n, m;
    cin>>n>>m;
    for (int i = 0; i < m; i++) {
        int x, y, z;cin>>x>>y>>z;
        q.push(edge(x, y, z));
    }
    int t = Kruskal(n);
    if (t == -1) cout<<"orz";
    else cout<<t;
    return 0;
}
```

力扣1584. 连接所有点的最小费用

给你一个`points` 数组，表示 2D 平面上的一些点，其中 `points[i] = [xi, yi]` 。

连接点 `[xi, yi]` 和点 `[xj, yj]` 的费用为它们之间的 **曼哈顿距离** ：`|xi - xj| + |yi - yj|` ，其中 `|val|` 表示 `val` 的绝对值。

请你返回将所有点连接的最小总费用。只有任意两点之间 **有且仅有** 一条简单路径时，才认为所有点都已连接。

> Prim算法
> 

```cpp
class Solution {
    private:
        vector<vector<pair<int,int>>> mp;
        vector<bool> flag;

        int prim(int start) {
            int sum = 0;
            priority_queue<
                tuple<int,int,int>,
                vector<tuple<int,int,int>>,
                greater<tuple<int,int,int>>
            > q;
            
            flag[start] = true;
            for (pair<int,int>& t:mp[start]) {
                int v = t.first, w = t.second;
                q.push(make_tuple(w, start, v));
            }

            while (!q.empty()) {
                auto t = q.top();q.pop();
                int w = get<0>(t), u = get<1>(t), v = get<2>(t);
                if (flag[v]) continue;
                flag[v] = true;
                sum += w;
                for (pair<int,int>& t : mp[v]) {
                    int w2 = t.second, y = t.first;
                    if (!flag[y]) q.push(make_tuple(w2, v, y));
                }
            }

            if (find(flag.begin(), flag.end(), false) != flag.end()) return 0;
            return sum;
        }
    public:
        int minCostConnectPoints(vector<vector<int>>& points) {
            int n = points.size();
            mp = vector<vector<pair<int,int>>>(n);
            flag = vector<bool>(n, false);
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    int x1 = points[i][0], y1 = points[i][1];
                    int x2 = points[j][0], y2 = points[j][1];
                    int w = abs(x1 - x2) + abs(y1 - y2);
                    mp[i].push_back({j, w});
                    mp[j].push_back({i, w}); 
                }
            }
            return prim(0);
        }
    };
```

```cpp
class Solution {
    private:
        struct Edge {
            int u,v,w;
            Edge(int a, int b, int c) : u(a), v(b), w(c) {}
        };
        vector<Edge> mp;

        class UnionFold {
            private:
                vector<int> parent;
                vector<int> size;
            public:
                UnionFold(int n) : parent(n+1), size(n+1, 1) {
                    for (int i = 0; i <= n; i++)
                        parent[i] = i;
                }

                bool merge(int p, int q) {
                    int rootP = find(p);
                    int rootQ = find(q);
                    if (rootP == rootQ) return false;

                    if (size[rootQ] > size[rootP]) {
                        parent[rootP] = rootQ;
                        size[rootQ] += size[rootP];
                    } else {
                        parent[rootQ] = rootP;
                        size[rootP] += size[rootQ];
                    }
                    return true;
                }

                int find(int x) {
                    while (x != parent[x]) {
                        parent[x] = parent[parent[x]];
                        x = parent[x];
                    }
                    return x;
                }
        };

        int Kruskal(int n) {
            int sum = 0, cnt = 0;
            sort(mp.begin(), mp.end(), [](auto& a,auto& b){
                return a.w < b.w;
            });
            UnionFold uf(n);
            for (Edge& t:mp){
                if(uf.merge(t.u, t.v)) {
                   sum += t.w;
                   cnt++;
                   if (cnt == n-1) break; 
                }
            }
            return cnt < n-1 ? 0 : sum;
        }
    public:
        int minCostConnectPoints(vector<vector<int>>& points) {
            int n = points.size();
            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    int x1 = points[i][0], y1 = points[i][1];
                    int x2 = points[j][0], y2 = points[j][1];
                    int w = abs(x1 - x2) + abs(y1 - y2);
                    mp.push_back(Edge(i, j, w));
                    mp.push_back(Edge(j, i, w));
                }
            }
            return Kruskal(n);
        }
    };
```

洛谷**P4779 【模板】单源最短路径（标准版）**

给定一个 *n* 个点，*m* 条有向边的带非负权图，请你计算从 *s* 出发，到每个点的距离。

数据保证你能从 *s* 出发到任意点。

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
struct State {
    int node,dis;   // 当前节点,前往节点的距离[权值]
    State(int n, int d) : node(n), dis(d) {}
    // 小顶堆[确保升序第一个最小]
    bool operator<(const State& other) const {
        return dis > other.dis;
    }
};
vector<vector<pair<int,int>>> mp;
// Dijkstra:计算从原点开始每个点的带权距离
// 使用BFS的方法,即Dijkstra是BFS的加权版本
vector<int> Dijkstra(int n, int start) {
    vector<int> dis(n+1, -1);
    priority_queue<State> pq;
    pq.push(State(start, 0));
    while (!pq.empty()) {
        State t = pq.top();pq.pop();
        int v = t.node, w = t.dis;
        // 距离已更新过,就跳过
        if (dis[v] != -1) continue;
        dis[v] = w;
        for (pair<int,int>& p : mp[v]) {
            int y = p.first, w2 = w + p.second;
            if (dis[y] == -1) pq.push(State(y, w2));
        }
    }
    return dis;
}
int main() {
    IOS;
    int n, m, s;cin>>n>>m>>s;
    mp = vector<vector<pair<int,int>>>(n+1);
    for (int i = 0; i < m; i++) {
        int u, v, w;cin>>u>>v>>w;
        mp[u].push_back({v, w});
    }
    vector<int> res = Dijkstra(n, 1);
    for (int i = 1; i <= n; i++)
        cout<<res[i]<<(i != n ? " " : "");
    return 0;
}
```

力扣743. 网络延迟时间

有 `n` 个网络节点，标记为 `1` 到 `n`。

给你一个列表 `times`，表示信号经过 **有向** 边的传递时间。 `times[i] = (ui, vi, wi)`，其中 `ui` 是源节点，`vi` 是目标节点， `wi` 是一个信号从源节点传递到目标节点的时间。

现在，从某个节点 `K` 发出一个信号。需要多久才能使所有节点都收到信号？如果不能使所有节点收到信号，返回 `-1` 。

```cpp
class Solution {
    private:
        vector<vector<pair<int,int>>> mp;
        struct State {
            int node, dis;
            State(int n, int d) : node(n), dis(d) {}
            bool operator<(const State& other) const {
                return dis > other.dis;
            }
        };
        int Dijkstra(int n, int start) {
            vector<int> dis(n+1, -1);
            priority_queue<State> q;
            q.push(State(start, 0));
            while (!q.empty()) {
                State t = q.top();
                int v = t.node, w = t.dis;
                q.pop();
                if (dis[v] != -1) continue;
                dis[v] = w;
                for (pair<int,int>& t : mp[v]) {
                    int y = t.first, w2 = w + t.second;
                    if (dis[y] == -1) q.push(State(y, w2));
                }
            }
            if (find(dis.begin()+1, dis.end(), -1) != dis.end()) return -1;
            else return *max_element(dis.begin()+1, dis.end());
        }
    public:
        int networkDelayTime(vector<vector<int>>& times, int n, int k) {
            mp = vector<vector<pair<int,int>>>(n+1);
            for(vector<int>& t : times)
                mp[t[0]].push_back({t[1], t[2]});
            return Dijkstra(n, k);
        }
    };
```

力扣1631. 最小体力消耗路径

你准备参加一场远足活动。给你一个二维 `rows x columns` 的地图 `heights` ，其中 `heights[row][col]` 表示格子 `(row, col)` 的高度。一开始你在最左上角的格子 `(0, 0)` ，且你希望去最右下角的格子 `(rows-1, columns-1)` （注意下标从 **0** 开始编号）。你每次可以往 **上**，**下**，**左**，**右** 四个方向之一移动，你想要找到耗费 **体力** 最小的一条路径。

一条路径耗费的 **体力值** 是路径上相邻格子之间 **高度差绝对值** 的 **最大值** 决定的。

请你返回从左上角走到右下角的最小 **体力消耗值** 。

```cpp
class Solution {
    private:
        vector<vector<pair<int,int>>> mp;
        struct State {
            int row, col, dis;
            State(int r, int c, int d) : row(r), col(c), dis(d) {}
            bool operator<(const State& other) const {
                return dis > other.dis;
            }
        };
        int Dijkstra(vector<vector<int>>& matrix) {
            int m = matrix.size(), n = matrix[0].size();
            vector<vector<int>> dis(m, vector<int>(n, -1));
            priority_queue<State> q;
            q.emplace(0, 0, 0);
            while (!q.empty()) {
                State t = q.top();
                int r = t.row, c = t.col, w = t.dis;
                q.pop();
                if (dis[r][c] != -1) continue;
                dis[r][c] = w;
                if (r == m-1 && c == n-1) return w;
                for (pair<int,int>& t : adj(matrix, r, c)) {
                    int x = t.first, y= t.second;
                    int nw = max(w, abs(matrix[x][y] - matrix[r][c]));
                    if (dis[x][y] != -1) continue;
                    q.emplace(x, y, nw);
                }
            }
            return -1;
        }
        vector<pair<int,int>> adj(vector<vector<int>>& matrix, int x, int y) {
            vector<int> dx = {-1, 0, 1, 0}, dy = {0, 1, 0, -1};
            int m = matrix.size(), n = matrix[0].size();
            vector<pair<int,int>> res;
            for (int i = 0; i < 4; i++) {
                int a = x + dx[i], b = y + dy[i];
                if (a < 0 || b < 0 || a >= m || b >= n) continue;
                res.push_back({a, b});
            }
            return res;
        }
    public:
        int minimumEffortPath(vector<vector<int>>& heights) {
            return Dijkstra(heights);
        }
    };
```
