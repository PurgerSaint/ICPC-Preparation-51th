### 力扣**115. 不同的子序列【复习】**

给你两个字符串 `s` ****和 `t` ，统计并返回在 `s` 的 **子序列** 中 `t` 出现的个数。

测试用例保证结果在 32 位有符号整数范围内。

```cpp
class Solution {
private:
    vector<vector<int>> lt;
public:
    int numDistinct(string s, string t) {
        int n = s.length(), m = t.length();
        lt = vector<vector<int>>(n+1, vector<int>(m+1, -1));
        return dp(s, 0, t, 0);
    }
private:
    int dp(string& s, int i, string& t, int j) {
        if (j == t.length()) return 1;
        if (s.length() - i < t.length() - j) return 0;
        if (lt[i][j] != -1) return lt[i][j];
        int res = 0;
        if (s[i] == t[j])
            // 继承当前子串方案 或 放弃当前匹配的子串
            res = dp(s, i+1, t, j+1) + dp(s, i+1, t, j);
        else res = dp(s, i+1, t, j);
        lt[i][j] = res;
        return res;
    }
};
```

### 力扣**139. 单词拆分【复习】**

给你一个字符串 `s` 和一个字符串列表 `wordDict` 作为字典。如果可以利用字典中出现的一个或多个单词拼接出 `s` 则返回 `true`。

**注意：**不要求字典中出现的单词全部都使用，并且字典中的单词可以重复使用。

```cpp
class Solution {
private:
    unordered_set<string> wordDict;
    unordered_set<string> nope;
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        this->wordDict = unordered_set<string>(wordDict.begin(), wordDict.end());
        return dp(s, 0);
    }
private:
    bool dp(string& s, int i) {
        if (i == s.length()) return true;
        if (nope.count(s.substr(i))) return false;
        for (int len = 1; i + len <= s.length(); len++) {
            if (wordDict.count(s.substr(i, len))) {
                if (dp(s, i + len))
                    return true;
            }
        }
        nope.insert(s.substr(i));
        return false;
    }
};
```

### 力扣**140. 单词拆分 II【复习】**

给定一个字符串 `s` 和一个字符串字典 `wordDict` ，在字符串 `s` 中增加空格来构建一个句子，使得句子中所有的单词都在词典中。**以任意顺序** 返回所有这些可能的句子。

**注意：**词典中的同一个单词可能在分段中被重复使用多次。

```cpp
class Solution {
private:
    unordered_set<string> wordDict;
    unordered_set<string> nope;
    vector<vector<string>> lt;
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        lt = vector<vector<string>>(s.length());
        this->wordDict = unordered_set<string>(wordDict.begin(), wordDict.end());
        return dp(s, 0);
    }
private:
    vector<string> dp(string& s, int i) {
        vector<string> res;
        if (i == s.length()) {
            res.push_back("");
            return res;
        }
        if (!lt[i].empty()) return lt[i];
        for (const string& word : wordDict) {
            int len = word.size();
            if (s.substr(i, len) == word) {
                for (string& t : dp(s, i + len))
                    if (t.empty()) res.push_back(word);
                    else res.push_back(word + " " + t);
            }
        }
        lt[i] = res;
        return res;
    }
};
```

### 力扣72.编辑距离【复习】

给你两个单词 `word1` 和 `word2`， *请返回将 `word1` 转换成 `word2` 所使用的最少操作数*  。

你可以对一个单词进行如下三种操作：

- 插入一个字符
- 删除一个字符
- 替换一个字符

```cpp
class Solution {
public:
    int minDistance(string s, string t) {
        int n = s.size(), m = t.size();
        vector<vector<int>> lt(n+1, vector<int>(m+1, 0));
        for (int i = 1; i <= n; i++)
            lt[i][0] = i;
        for (int j = 1; j <= m; j++)
            lt[0][j] = j;
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++)
                if (s[i-1] == t[j-1])
                    lt[i][j] = lt[i-1][j-1];
                else                    // 删除 替换 插入
                    lt[i][j] = 1 + min3(lt[i-1][j], lt[i-1][j-1], lt[i][j-1]);
        return lt[n][m];
    }
private:
    int min3(int a, int b, int c) {
        return min(a, min(b, c));
    }
};
```

### 力扣**516. 最长回文子序列**

给你一个字符串 `s` ，找出其中最长的回文子序列，并返回该序列的长度。

子序列定义为：不改变剩余字符顺序的情况下，删除某些字符或者不删除任何字符形成的一个序列。

> 定义lt[i][j]为字符串s中从s[i]到s[j]的最长回文子序列长度
如果s[i]==s[j]，那就lt[i-1][j-1] + 1
不对，这与之前的定义冲突
详细拆解本题，如果s[i]==s[j]，那就是从s[i+1]到s[j-1]的子序列长度加2
这才是动态规划拆解出的子问题
如果s[i]!=s[j]，假设s[i]是字符串内的，就考虑lt[i][j-1]；
还可以假设s[j]是回文字符串内的，就考虑lt[i+1][j]
此时取得这俩最大值即可，此乃区间DP，而非LCS的前缀DP
> 

```cpp
class Solution {
    private:
        vector<vector<int>> lt;
    public:
        int longestPalindromeSubseq(string s) {
            int n = s.length();
            lt = vector<vector<int>>(n, vector<int>(n, 0));
            // 单个字符作为子序列必是回文子串
            for (int i = 0; i < n; i++)
                lt[i][i] = 1;
            // 从后往前，避免后续的没算
            for (int i = n - 1; i >= 0; i--)
                for (int j = i + 1; j < n; j++)
                    if (s[i] == s[j])
                        lt[i][j] = lt[i + 1][j - 1] + 2;
                    else
                        lt[i][j] = max(lt[i+1][j], lt[i][j-1]);
            // 返回{0, ... ,n-1}的回文子串即可
            return lt[0][n-1];
        }
    };
```

### 力扣**1312. 让字符串成为回文串的最少插入次数**

给你一个字符串 `s` ，每一次操作你都可以在字符串的任意位置插入任意字符。

请你返回让 `s` 成为回文串的 **最少操作次数** 。

「回文串」是正读和反读都相同的字符串。

> 根据题意，可以定义lt[i][j]为s{i, ... ,j}成为回文串的最小次数
如果s[i]==s[j]，则lt[i][j]等于lt[i+1][j-1]
如果s[i]!=s[j]，则lt[i][j]有两种情况，一个是插入s[i]，再考虑s{i+1, ... ,j}成为回文串的次数，即lt[i+1][j]+1；
另一种情况是插入s[j]，再考虑s{i, ... ,j-1}成为回文串的次数，即lt[i][j-1]+1;
题目要求最小次数，此时取这两种情况的最小值即可
由于保证j永远大于i，我们只需遍历右上角，取得lt[0][n-1]即可
同时没有特殊条件，二维数组全部初始化为0即可
> 

```cpp
class Solution {
public:
    int minInsertions(string s) {
        int n = s.length();
        vector<vector<int>> lt(n, vector<int>(n, 0));
        for (int i = n - 1; i >= 0; i--)
            for (int j = i + 1; j < n; j++)
                if (s[i] == s[j])
                    lt[i][j] = lt[i+1][j-1];
                else lt[i][j] = min(lt[i+1][j], lt[i][j-1]) + 1;
        return lt[0][n-1];
    }
};
```

### 力扣474. 一和零

给你一个二进制字符串数组 `strs` 和两个整数 `m` 和 `n` 。

请你找出并返回 `strs` 的最大子集的长度，该子集中 **最多** 有 `m` 个 `0` 和 `n` 个 `1` 。

如果 `x` 的所有元素也是 `y` 的元素，集合 `x` 是集合 `y` 的 **子集** 。

> 根据题意，先定义lt[k][i][j]为选择k个元素后，要求i个0和j个1能组成的子集个数
如果当前元素被选择，则选择lt[k-1][i-num0][j-num1]+1(选择当前字符串后需要的个数)和lt[k-1][i][j](不选择当前字符串需要的个数)的最大值即可
否则就继承lt[k-1][i][j]
> 

```cpp
class Solution {
public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        vector<vector<vector<int>>> lt(strs.size()+1, vector<vector<int>>(m+1, vector<int>(n+1, 0)));
        for (int k = 1; k <= strs.size(); k++) {
            string& s = strs[k-1];
            int num0 = count(s.begin(), s.end(), '0');
            int t = s.length(), num1 = t - num0;
            for (int i = 0; i <= m; i++)
                for (int j = 0; j <= n; j++) {
                    if (i - num0 >= 0 && j - num1 >= 0)
                        lt[k][i][j] = max(
                            lt[k - 1][i - num0][j - num1] + 1,
                            lt[k - 1][i][j]
                        );
                    else lt[k][i][j] = lt[k - 1][i][j];
                }
        }
        return lt[strs.size()][m][n];
    }
};
```

> 同时0-1背包也可以进行二维压缩
我们仅需定义lt[i][j]为使用i个0和j个1所需要的子集个数
然后倒序遍历，寻找不使用当前元素(lt[i][j])和使用当前元素(lt[i-num0][j-num1]+1)的最大值即可。
> 

```cpp
class Solution {
public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        vector<vector<int>> lt(m+1, vector<int>(n+1, 0));
        for (string& s : strs) {
            int num0 = count(s.begin(), s.end(), '0');
            int t = s.length(), num1 = t - num0;
            for (int i = m; i >= num0; i--)
                for (int j = n; j >= num1; j--) {
                    lt[i][j] = max(lt[i][j], lt[i-num0][j-num1] + 1);
                }
        }
        return lt[m][n];
    }
};
```

### 力扣1262. 可被三整除的最大和

给你一个整数数组 `nums`，请你找出并返回能被三整除的元素 **最大和**。

> 如果按题意定义dp[i]为选择i个数后能被三整除的最大和
而后怎么从dp[i-1]转移到dp[i]呢？毕竟我还得加一个选择,代表选择的是哪个数
不对，这样会状态重复，且子问题不明确
那如果根据每个数的余数作为转移，毕竟题目最终也是根据每个数的余数作为转移的
即定义dp[i][j]为选择i个数后，余数为j的最大和
遍历每个数，根据余数更新对应选择（选择当前数或不选择），最后返回dp[n][0]即可
> 

```cpp
class Solution {
    public:
        int maxSumDivThree(vector<int>& nums) {
            int n = nums.size();
            // 定义状态lt[i][j]表示选i个数、余数为j的最大和
            vector<vector<int>> lt(n+1, vector<int>(3, INT_MIN));
            lt[0][0] = 0;
    
            for (int i = 1; i <= n; i++) {
                int& t = nums[i - 1];
                if (t % 3 == 0) {
                    lt[i][0] = lt[i - 1][0] + t;
                    // 余数为1和2的最大和加上余数为0的数余数不变
                    lt[i][1] = lt[i - 1][1] + t;
                    lt[i][2] = lt[i - 1][2] + t;
                // 后续更新确保余数一定为对应的{0,1,2}
                } else if (t % 3 == 1) {
                    lt[i][0] = max(lt[i - 1][2] + t, lt[i - 1][0]);
                    lt[i][1] = max(lt[i - 1][0] + t, lt[i - 1][1]);
                    lt[i][2] = max(lt[i - 1][1] + t, lt[i - 1][2]);
                } else {
                    lt[i][0] = max(lt[i - 1][1] + t, lt[i - 1][0]);
                    lt[i][1] = max(lt[i - 1][2] + t, lt[i - 1][1]);
                    lt[i][2] = max(lt[i - 1][0] + t, lt[i - 1][2]);
                }
            }
            return lt[n][0];
        }
    };
```

### 力扣1235. 规划兼职工作

你打算利用空闲时间来做兼职工作赚些零花钱。

这里有 `n` 份兼职工作，每份工作预计从 `startTime[i]` 开始到 `endTime[i]` 结束，报酬为 `profit[i]`。

给你一份兼职工作表，包含开始时间 `startTime`，结束时间 `endTime` 和预计报酬 `profit` 三个数组，请你计算并返回可以获得的最大报酬。

注意，时间上出现重叠的 2 份工作不能同时进行。

如果你选择的工作在时间 `X` 结束，那么你可以立刻进行在时间 `X` 开始的下一份工作。

> 按示例一的数据进行模拟选择，首先按会议结束时间进行排序（贪心会议问题）
如下：{1,3,50}、{2,4,10}、{3,5,40}、{3,6,70}
如果选择前一份工作，选择的是第一份，报酬共50
如果选择前两份工作，选择的是第一份，报酬共50
如果选择前三份工作，选择的是第一份和第三份，报酬共50+40=90
如果选择前四份工作，选择的是第一份和第四份，报酬共50+70=120
所以不妨定义lt[i]为选择前i份工作所能获得的最大报酬
同时也可以从lt[i-1](不选该工作)和lt[j]+work.profit(选择该工作前能选择的最大工作报酬)中选择一个最大值推导至lt[i]
> 

```cpp
class Solution {
private:
    struct Work {
        int start, end, profit;
        Work(int s, int e, int p) : start(s), end(e), profit(p) {}
        bool operator<(const Work& other) const {
            return end < other.end;
        }
    };
public:
    int jobScheduling(vector<int>& startTime, vector<int>& endTime, vector<int>& profit) {
        int n = startTime.size();
        vector<Work> work;
        for (int i = 0; i < n; i++)
            work.push_back(Work(startTime[i], endTime[i], profit[i]));
        sort(work.begin(), work.end());
        vector<int> lt(n+1, 0);
        for (int i = 1; i <= n; i++) {
            int j = upper_bound(work.begin(), work.begin() + i - 1, work[i - 1].start, [](int val,const Work& other){
                return val < other.end;
            }) - work.begin();
            lt[i] = max(lt[i - 1], lt[j] + work[i - 1].profit);
        }
        return lt[n];
    }
};
```

### 力扣**3180. 执行操作可获得的最大总奖励 I**

给你一个整数数组 `rewardValues`，长度为 `n`，代表奖励的值。

最初，你的总奖励 `x` 为 0，所有下标都是 **未标记** 的。你可以执行以下操作 **任意次** ：

- 从区间 `[0, n - 1]` 中选择一个 **未标记** 的下标 `i`。
- 如果 `rewardValues[i]` **大于** 你当前的总奖励 `x`，则将 `rewardValues[i]` 加到 `x` 上（即 `x = x + rewardValues[i]`），并 **标记** 下标 `i`。

以整数形式返回执行最优操作能够获得的 **最大** **总奖励。

> 如果定义lt[i]为选择前i个物品所获得的最大奖励
那此时依据题目要求，只能选择比总和大的数，无法从lt[i]推导至lt[i]
则此时可以改变思路，选择lt[i][m]，定义为选择i个物品是否能获得m的奖励
并且所获结果的最大值一定包括数组最大值，因为最优解的最后一步一定选的是数组最大值
且此时可知选择最大值前的最大奖励为最大值减一，加上最大值，可知奖励最大只能取得最大值两倍减一
> 

```cpp
class Solution {
public:
    int maxTotalReward(vector<int>& rewardValues) {
        int n = rewardValues.size();
        sort(rewardValues.begin(), rewardValues.end());
        int m = rewardValues[n-1];
        vector<vector<bool>> lt(n+1, vector<bool>(2*m, false));
        lt[0][0] = true;

        for (int i = 1; i <= n; i++) {
            int& curVal = rewardValues[i-1];
            for (int x = 0; x <= 2 * m - 1; x++) {
                // 前一步总和不为0 且 当前值大于前一步总和
                if (x - curVal >= 0 && curVal > x - curVal) {
                    // 选择当前元素
                    bool add = lt[i-1][x - curVal];
                    // 不选当前元素
                    bool notadd = lt[i-1][x];
                    lt[i][x] = add || notadd;
                } else {
                    lt[i][x] = lt[i-1][x];
                }
            }
        }
        for (int i = 2 * m - 1; i >= 0; i--)
            if (lt[n][i]) return i;
        return 0;
    }
};
```
