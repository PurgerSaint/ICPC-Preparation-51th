### 力扣509.斐波那契数

**斐波那契数** （通常用 `F(n)` 表示）形成的序列称为 **斐波那契数列** 。该数列由 `0` 和 `1` 开始，后面的每一项数字都是前面两项数字的和。也就是：

$F(0) = 0，F(1) = 1，F(n) = F(n - 1) + F(n - 2)，其中 n > 1$

给定 `n` ，请计算 `F(n)` 。

> 计算斐波那列数列的关键信息知道前两项斐波那列的值
就跟之前的图论DFS一样，建立一个决策树，自顶向下递归计算每一个计算的值
例如fib(4)，你必须要先计算fib(3)和fib(2)，然后就是fib(1)和fib(2)与fib(1)和fib(0)
这样通过递归便能很快计算出一个版本
> 

```cpp
class Solution {
public:
    int fib(int n) {
        return n == 0 ? 0 : (n == 1 ? 1 : fib(n-1) + fib(n-2));
    }
};
```

> 但是吧，这样纯粹的递归虽能解决问题，但是会耗费大量时间来遍历重复的节点
例如对于fib(3)和fib(4)，都要重复计算fib(2)
那么我们是不是可以通过DFS剪枝，即记忆化搜索来解决
即对于fib(n)，我们只要知道fib(n-1)和fib(n-2)便自然知道fib(n)
同时也是对于fib(n)这个大问题，缩小到fib(n-1)和fib(n-2)这两个小问题中
> 

```cpp
class Solution {
public:
    int fib(int n) {
        vector<int> f(n+1, 0);
        for (int i = 0; i <= n; i++)
            if (i == 0) f[i] = 0;
            else if (i == 1) f[i] = 1;
            else f[i] = f[i-1] + f[i-2];
        return f[n];
    }
};
```

### 力扣**322. 零钱兑换**

给你一个整数数组 `coins` ，表示不同面额的硬币；以及一个整数 `amount` ，表示总金额。

计算并返回可以凑成总金额所需的 **最少的硬币个数** 。如果没有任何一种硬币组合能组成总金额，返回 `-1` 。

你可以认为每种硬币的数量是无限的。

> 对于本题来说，假定我有三种硬币[1, 2, 5]要凑齐11元
初始我可以选择1元硬币、2元硬币或5元硬币
这样问题就变成了凑齐10元、9元或6元需要的最少硬币个数
就有点动态规划的味道了，由一个大问题分成诸多个小问题
且这些小问题相互独立，即凑齐10元所需的硬币数不会影响凑齐9元所需的硬币数
同时也是做出一个递归树，自上而下的
那么就可以简单地写出一个简单的动态规划递归版本
> 

```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        return dp(coins, amount);
    }
private:
    int dp(vector<int>& coins, int amount) {
        if (amount == 0)    return 0;
        if (amount < 0)     return -1;
        int ans = INT_MAX;
        for (int& coin : coins) {
            int res = dp(coins, amount - coin);
            if (res == -1) continue;
            ans = min(res + 1, ans);
        }
        return ans == INT_MAX ? -1 : ans;
    }
};
```

> 等会，决策树，最短路径，貌似我可以使用BFS来解决
> 

```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        return dp(coins, amount);
    }
private:
    int bfs(vector<int>& coins, int amount) {
        if (amount == 0)    return 0;
        int ans = INT_MAX;
        vector<bool> flag(amount+1, false);
        queue<pair<int,int>> q;
        q.push({amount, 0});
        flag[amount] = true;

        while (!q.empty()) {
            auto t = q.front();
            int cost = t.first, time = t.second;
            q.pop();
            if (cost == 0) return time;
            for (int& p : coins) {
                int u = cost - p;
                if (u < 0) continue;
                if (!flag[u]) {
                    flag[u] = true;
                    q.push({u, time + 1});
                }
            }
        }
        return -1;
    }
};
```

> 既然是决策树，递归做法必有重复访问
那么就可以通过剪枝，即记忆化搜索来解决重复访问
> 

```cpp
class Solution {
private:
    vector<int> cost;
public:
    int coinChange(vector<int>& coins, int amount) {
        // 不能初始化为-1:因为无解情况下为-1,会冲突
        cost = vector<int>(amount+1, -2);
        return dp(coins, amount);
    }
private:
    int dp(vector<int>& coins, int amount) {
        if (amount == 0) return 0;
        if (amount < 0) return -1;
        // 判重时机:无解和有解之后
        if (cost[amount] != -2) return cost[amount];
        int res = INT_MAX;
        for (int& coin : coins) {
            int ans = dp(coins, amount - coin);
            if (ans == -1) continue;
            res = min(res, ans + 1);
        }
        cost[amount] = res == INT_MAX ? -1 : res;
        return cost[amount];
    }
};
```

### 力扣300.最长递增子序列

给你一个整数数组 `nums` ，找到其中最长严格递增子序列的长度。

**子序列** 是由数组派生而来的序列，删除（或不删除）数组中的元素而不改变其余元素的顺序。例如，`[3,6,2,7]` 是数组 `[0,3,1,6,2,2,7]` 的子序列。

> 最长递增子序列？那我用DFS找出所有按原顺序的序列，然后找到那个递增的最长序列不就得了
> 

```cpp
class Solution {
private:
    vector<vector<int>> res;
    void dfs(vector<int>& nums, vector<int>& path, int i) {
        int n = nums.size();
        if (i >= n) return;
        for (int j = i + 1; j < n; j++) {
            path.push_back(nums[j]);
            res.push_back(path);
            dfs(nums, path, j);
            path.pop_back();
        }
    }
public:
    int lengthOfLIS(vector<int>& nums) {
        vector<int> path = {};
        res.push_back(path);
        dfs(nums, path, -1);
        sort(res.begin(), res.end(), [](vector<int>& a, vector<int>& b){
            return a.size() > b.size();
        });
        for (vector<int>& p : res) {
            if (p.empty()) continue;
            int cnt = 1, n = p.size();
            for (int i = 1; i < n; i++)
                if (p[i] > p[i-1]) cnt++;
                else break;
            if (cnt < n) continue;
            return cnt;
        }
        return 1;
    }
};
```

> 但是不行，DFS对于大量数据会超出内存限制。
等等，如果假设最后一个值就是最长递增子序列的最后一个元素
那我岂不是可以做一个决策树，找到值比他小，然后计算路径即可
使用BFS
> 

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> sz(n, 1);
        queue<pair<int,int>> q;
        q.push({n-1, 1});
        int res = INT_MIN;
        while (!q.empty()) {
            auto t = q.front();
            q.pop();
            int i = t.first, u = t.second;
            if (sz[i] != 0) continue;
            sz[i] = u;
            for (int j = 0; j < i; j++) {
                if (nums[j] < nums[i]) {
                    q.push({j, u + 1});
                }
            }
        }
        return *max_element(sz.begin(), sz.end());
    }
};
```

> BFS也不行，假设最后一个元素是递增子序列最后一个元素是不一定成立的
要是往前对每一个元素都假设为递增子序列最后一个元素计算路径
但也不行，这样时间复杂度会很高，非常容易超时
等等，假设元素n-1是递增子序列的最后一个元素
那对于长度n数组的最长递增子序列长度就是长度n-1数组的最长长度加一
这样就是动态规划问题，把n大小的最长长度缩小到n-1大小的最长长度
使用DP+记忆化试试
> 

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n, 1);
        for (int i = 0; i < n; i++)
            // 找找他前面有没有元素可以构成递增子序列的
            for (int j = 0; j < i; j++)
                if (nums[j] < nums[i])
                    dp[i] = max(dp[i], dp[j] + 1);
        return *max_element(dp.begin(), dp.end());
    }
};
```

> DP+记忆化成功了
之前用DFS，我们是为了找找那个真正的递增子序列
那如果我们遍历数组，使用牌堆的想法来构造一个长度一样的子序列
保证这个子序列始终递增，比最大值还大就插入，没有就找到第一个比他大的进行替换即可
比他大使用二分查找即可
> 

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> p(n, 0);
        int plies = 0;

        for (int& poker : nums) {
            int left = 0, right = plies;
            while (left < right) {
                int mid = left + ((right - left) >> 1);
                if (p[mid] >= poker) right = mid;
                else left = mid + 1;
            }
            if (left == plies) plies++;
            p[left] = poker;
        }
        return plies;
    }
};
```

> 还有一个更快的方法，就是使用库函数的牌堆法优化
> 

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> p;

        for (int& poker : nums) {
            if (p.empty() || poker > p.back())
                p.push_back(poker);
            else {
                auto it = lower_bound(p.begin(), p.end(), poker);
                *it = poker;
            }
        }
        return p.size();
    }
};
```

### 力扣**354. 俄罗斯套娃信封问题**

给你一个二维整数数组 `envelopes` ，其中 `envelopes[i] = [wi, hi]` ，表示第 `i` 个信封的宽度和高度。

当另一个信封的宽度和高度都比这个信封大的时候，这个信封就可以放进另一个信封里，如同俄罗斯套娃一样。

请计算 **最多能有多少个** 信封能组成一组“俄罗斯套娃”信封（即可以把一个信封放到另一个信封里面）。

**注意**：不允许旋转信封。

> A信封的宽高都比B信封大时，B信封才能放入A信封
A信封内的数量不就是B信封内的数量加一嘛，动态规划找到了子问题
同时也是寻找一个长度和高度的最长递增子序列长度
那么直接对长度进行排序，寻找一个最长的递增序列即可
不对，虽然保证长度递增，可是怎么保证高度也同样递增呢？
好问题，那长度递增，高底递减不就可以了
此时对于高度这一序列，相同长度的高度依次降序
那么就直接单拎高度出来，高度序列中存在的子序列必定是长度和高度的子序列
这样就完成了二维LIS的解决
先降维，然后套用LIS的问题代码解答即可
> 

```cpp
class Solution {
public:
    int maxEnvelopes(vector<vector<int>>& envelopes) {
        sort(envelopes.begin(), envelopes.end(), [](vector<int>& a, vector<int>& b){
            return a[0] == b[0] ? a[1] > b[1] : a[0] < b[0];
        });
        int n = envelopes.size();
        vector<int> height(n);
        for (int i = 0; i < n; i++)
            height[i] = envelopes[i][1];
        return LIS(height);
    }
private:
    int LIS(vector<int>& height) {
        vector<int> top(height.size());
        int plies = 0;
        for (int& poker : height) {
            int left = 0, right = plies;
            while (left < right) {
                int mid = left + ((right - left) >> 1);
                if (poker > top[mid]) left = mid + 1;
                else right = mid;
            }
            if (left == plies) plies++;
            top[left] = poker;
        }
        return plies;
    }
};
```

### 力扣**931. 下降路径最小和**

给你一个 `n x n` 的 **方形** 整数数组 `matrix` ，请你找出并返回通过 `matrix` 的**下降路径** **的 **最小和** 。

**下降路径** 可以从第一行中的任何元素开始，并从每一行中选择一个元素。在下一行选择的元素和当前行所选元素最多相隔一列（即位于正下方或者沿对角线向左或者向右的第一个元素）。具体来说，位置 `(row, col)` 的下一个元素应当是 `(row + 1, col - 1)`、`(row + 1, col)` 或者 `(row + 1, col + 1)` 。

> 下降路径在第一行开始，最后一行结束
图的题目，是否可以用图论呢？决策树遍历会超时
等等，下降路径最小，那我岂不是计算到达最后一行的所有路径最小值即可
再等等，如果最后一行的某个值是该最小值路径，那么必定是倒数第二行某一个元素所在的最小路径和加上该最后一行某个值
也就是说，我只要知道上一行各元素的路径最小值，即可推出本行各元素的路径最小值
题目要求可以走左下、正下和右下，那直接以数组为路径最小值，计算各行各列的路径最小值即可，最后找到最后一行中最小的那个路径最小值即可
> 

```cpp
class Solution {
public:
    int minFallingPathSum(vector<vector<int>>& matrix) {
        int n = matrix.size(), m = matrix[0].size();
        vector<vector<int>> dp(n, vector<int>(m, 0));
        for (int i = 0; i < m; i++)
            dp[0][i] = matrix[0][i];
        for (int i = 1; i < n; i++)
            for (int j = 0; j < m; j++) {
                // 正上方
                dp[i][j] = matrix[i][j] + dp[i-1][j];
                // 左上
                if (j > 0) dp[i][j] = min(dp[i][j], matrix[i][j] + dp[i-1][j-1]);
                // 右上
                if (j < m - 1) dp[i][j] = min(dp[i][j], matrix[i][j] + dp[i-1][j+1]);
            }
        return *min_element(dp[n-1].begin(), dp[n-1].end());
    }
};
```

### 力扣**115. 不同的子序列**

给你两个字符串 `s` ****和 `t` ，统计并返回在 `s` 的 **子序列** 中 `t` 出现的个数。

测试用例保证结果在 32 位有符号整数范围内。

> 假设s[0,i]中出现了t[0,1]，那么是否可以找s[i+1,s.length()-1]中是否存在t[2,t.length()]即可
那么就是每个t[0]在s中组成子序列的次数总和就是题目答案
可以写出一个初步DP
> 

```cpp
class Solution {
public:
    int numDistinct(string s, string t) {
        return dp(s, 0, t, 0);
    }
private:
    int dp(string s, int i, string t, int j) {
        if (j == t.length()) return 1;
        if (s.length() - i < t.length() - j) return 0;
        int res = 0;
        for (int k = i; k < s.length(); k++)
            if (s[k] == t[j])
                res += dp(s, k + 1, t, j + 1);
        return res;
    }
};
```

> 可恶，初步DP超时，那加入记忆化
> 

```cpp
class Solution {
private:
    vector<vector<int>> lt;
public:
    int numDistinct(string s, string t) {
        lt = vector<vector<int>>(s.length(), vector<int>(t.length(), -1));
        return dp(s, 0, t, 0);
    }
private:
    int dp(string s, int i, string t, int j) {
        if (j == t.length()) return 1;
        if (s.length() - i < t.length() - j) return 0;
        if (lt[i][j] != -1) return lt[i][j];
        int res = 0;
        for (int k = i; k < s.length(); k++)
            if (s[k] == t[j])
                res += dp(s, k + 1, t, j + 1);
        lt[i][j] = res;
        return lt[i][j];
    }
};
```

> 糟糕，记忆化+DP依旧超时
看来是方法错误，刚刚是通过t，在s中逐步找到t的每一个匹配
可是要是通过s，找到每个t中字符的匹配，或许可行（已通过）
> 

```cpp
class Solution {
private:
    vector<vector<int>> lt;
public:
    int numDistinct(string s, string t) {
        lt = vector<vector<int>>(s.length(), vector<int>(t.length(), -1));
        return dp(s, 0, t, 0);
    }
private:
    int dp(string s, int i, string t, int j) {
        if (j == t.length()) return 1;
        if (s.length() - i < t.length() - j) return 0;
        if (lt[i][j] != -1) return lt[i][j];
        int res = 0;
        if (s[i] == t[j])
            res = dp(s, i+1, t, j+1) + dp(s, i+1, t, j);
        else res = dp(s, i+1, t, j);
        lt[i][j] = res;
        return lt[i][j];
    }
};
```

力扣**139. 单词拆分**

给你一个字符串 `s` 和一个字符串列表 `wordDict` 作为字典。如果可以利用字典中出现的一个或多个单词拼接出 `s` 则返回 `true`。

**注意：**不要求字典中出现的单词全部都使用，并且字典中的单词可以重复使用。

> 对于排列问题，我们可以使用DFS和回溯来寻找，是否存在一种排列方式
即对于字符串开头的每一段是否由字典中的一个单词组成
那么就可以接着找剩下字符串的开头是否由一个单词组成
> 

```cpp
class Solution {
private:
    bool found = false;
    deque<string> track;
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        backtrack(s, 0, wordDict);
        return found;
    }
private:
    void backtrack(string& s, int i, vector<string>& wordDict) {
        if (found) return;
        if (i == s.size()) {
            found = true;
            return;
        }
        for (string& word : wordDict) {
            int len = word.size();
            if (i + len <= s.size() && s.substr(i, len) == word) {
                track.push_back(word);
                backtrack(s, i + len, wordDict);
                track.pop_back();
            }
        }
    }
};
```

> 但是还是超时，看来只使用found进行递归截止并不行
还需要通过备忘录来进行剪枝优化
我们只用判断当前字符串是否不能被分解，即存在字典里找不到的值
那么体现就是在for循环后，found变量一直未被改变，则代表不可能（已通过）
> 

```cpp
class Solution {
private:
    bool found = false;
    deque<string> track;
    unordered_set<string> no;
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        backtrack(s, 0, wordDict);
        return found;
    }
private:
    void backtrack(string& s, int i, vector<string>& wordDict) {
        if (found) return;
        if (i == s.size()) {
            found = true;
            return;
        }
        string suf = s.substr(i);
        if (no.count(suf)) return;
        for (string& word : wordDict) {
            int len = word.size();
            if (i + len <= s.size() && s.substr(i, len) == word) {
                track.push_back(word);
                backtrack(s, i + len, wordDict);
                track.pop_back();
            }
        }
        if (!found) no.insert(suf);
    }
};
```

> 等会，之前好像分析过，如果s[0,k]是由某一个单词组成，那么我们只用慢慢用单词拼s[k+1,s.len]，就能完整拼出s
我们找到了子问题，即动态规划思想，可以初步尝试
> 

```cpp
class Solution {
```

> 直接DP不行，看来得使用优化DP，最好的方法就是使用备忘录（记忆化）
就像是DFS+回溯那，如果遇到了不可拆分的子串，我们直接放弃即可（已通过）
> 

```cpp
class Solution {
private:
    unordered_set<string> no;
public:
    bool wordBreak(string s, vector<string>& wordDict) {
        return dp(s, 0, wordDict);
    }
private:
    bool dp(string& s, int i, vector<string>& wordDict) {
        if (i == s.size()) return true;
        if (no.count(s.substr(i))) return false;
        for (string& word : wordDict) {
            int len = word.size();
            if (s.substr(i, len) == word) {
                if (dp(s, i + len, wordDict)) {
                    return true;
                }
            }
        }
        no.insert(s.substr(i));
        return false;
```

### 力扣**140. 单词拆分 II**

给定一个字符串 `s` 和一个字符串字典 `wordDict` ，在字符串 `s` 中增加空格来构建一个句子，使得句子中所有的单词都在词典中。**以任意顺序** 返回所有这些可能的句子。

**注意：**词典中的同一个单词可能在分段中被重复使用多次。

> 可以直接套用1的DFS+回溯代码
然后删除found，找出所有拼接的可能性
虽然代码时间复杂度很高，但是题目数据量小可以通过
> 

```cpp
class Solution {
private:
    deque<string> track;
    vector<string> res;
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        backtrack(s, 0, wordDict);
        return res;
    }
private:
    void backtrack(string& s, int i, vector<string>& wordDict) {
        if (i == s.size()) {
            string t = "";
            for (int i = 0; i < track.size(); i++) {
                t += track[i];
                if (i != track.size() - 1) t += " ";
            }
            res.push_back(t);
            return;
        }
        for (string& word : wordDict) {
            int len = word.size();
            if (i + len <= s.size() && s.substr(i, len) == word) {
                track.push_back(word);
                backtrack(s, i + len, wordDict);
                track.pop_back();
            }
        }
    }
};
```

> DP也可以套用代码，修改合法时进行拼接即可
同时记忆化修改为当前是否有拼接方式，即是否拼接过
> 

```cpp
class Solution {
private:
    unordered_set<string> wordDict;
    vector<vector<string>> lt;
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        // 对于每个s[i]是否存在一个拼接后缀
        lt = vector<vector<string>>(s.size());
        // 哈希寻找更快
        this->wordDict = unordered_set<string>(wordDict.begin(), wordDict.end());
        return dp(s, 0);
    }
private:
    vector<string> dp(string s, int i) {
        vector<string> res;
        // 代表当前有合法的拼接方式
        if (i == s.size()) {
            // 空字符串用于拼接
            res.push_back("");
            return res;
        }
        // 记忆化搜索（优化）
        if (!lt[i].empty()) return lt[i];
        for (int len = 1; i + len <= s.size(); len++) {
            string per = s.substr(i, len);
            if (wordDict.count(per)) {
                for (string t : dp(s, i + len)) {
                    // 当前单词是最后一个单词
                    if (t.empty()) res.push_back(per);
                    // 否则继续拼接
                    else res.push_back(per + " " + t);
                }
            }
        }
        lt[i] = res;
        return res;
    }
};
```

### 力扣72.编辑距离

给你两个单词 `word1` 和 `word2`， *请返回将 `word1` 转换成 `word2` 所使用的最少操作数*  。

你可以对一个单词进行如下三种操作：

- 插入一个字符
- 删除一个字符
- 替换一个字符

> 一个字符串需要经过多次操作能变成另一个字符串
假设第一个字符都相等，那么需要的次数就是剩下字符串的操作次数加一
子问题找到，那么就可以使用动态规划来解决本题了
> 

```cpp
class Solution {
private:
    vector<vector<int>> lt;
public:
    int minDistance(string word1, string word2) {
        int n = word1.size(), m = word2.size();
        lt = vector<vector<int>>(n, vector<int>(m, -1));
        return dp(word1, n - 1, word2, m - 1);
    }
private:
    int dp(string s, int i, string t, int j) {
        if (i == -1) return j + 1;
        if (j == -1) return i + 1;
        if (lt[i][j] != -1) return lt[i][j];
        if (s[i] == t[j]) lt[i][j] = dp(s, i-1, t, j-1);
        else lt[i][j] = min(
            // 插入:当前等效于在s后面插入了t[j]
            dp(s, i, t, j - 1),
            // 删除:当前等效于删除了s[i]
            min(dp(s, i - 1, t, j),
            // 替换:当前等效于将s[i]替换成t[j]
            dp(s, i - 1, t, j - 1))
        ) + 1;
        return lt[i][j];
    }
};
```

> 等会，我们之前找到了子问题，使用了DP
然后记忆化lt，lt[i][j]代表当前使得s[0,i]和t[0,j]相等需要的操作次数
等等，我们是不是可以从lt[0][0]直接推出到lt[n][m]
已知lt[i][j]代表当前使得s[0,i]和t[0,j]相等需要的操作次数
那么lt[i-1][j-1] + 1是将s[i-1]替换成t[j-1]后使得两字符串相等的操作次数
还有lt[i-1][j] + 1是将s[i-1]删除后使得两字符串相等的操作次数
以及lt[i][j-1] + 1是将t[i-1]插入至s后使得两字符串相等的操作次数
如果s[i]和t[j]相等，直接继承lt[i][j]即可
否则就取这三者最小值
这就是DP table
> 

```cpp
class Solution {
public:
    int minDistance(string s, string t) {
        int n = s.size(), m = t.size();
        vector<vector<int>> dp(n+1, vector<int>(m+1, 0));
        for (int i = 1; i <= n; i++)
            dp[i][0] = i;
        for (int j = 1; j <= m; j++)
            dp[0][j] = j;
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++)
                if (s[i-1] == t[j-1])
                    dp[i][j] = dp[i-1][j-1];
                else dp[i][j] = min(
                    dp[i-1][j],         // 删除
                    min(dp[i-1][j-1],   // 替换
                    dp[i][j-1]          // 替换
                )) + 1;
        return dp[n][m];
    }
};
```

### 力扣53.最大子数组和

给你一个整数数组 `nums` ，请你找出一个具有最大和的连续子数组（子数组最少包含一个元素），返回其最大和。

**子数组**是数组中的一个连续部分。

> 求得连续子数组中的最大值
连续，那么就可以使用滑动窗口
什么时候扩大窗口？加完右值就扩大
什么时候缩小窗口？窗口值非正时
什么时候更新最大值？加完右值时
> 

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int n = nums.size();
        int left = 0, right = 0;
        int winSum = 0, maxSum = INT_MIN;
        while (right < n) {
            winSum += nums[right];
            right++;

            maxSum = max(maxSum, winSum);

            while (winSum < 0) {
                winSum -= nums[left];
                left++;
            }
        }
        return maxSum;
    }
};
```

> 可以使用前缀和思路，一个连续数组的最大和必定是当前前缀和前去之前最小的前缀和
> 

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int n = nums.size();
        vector<int> sum(n+1, 0);
        for (int i = 1; i <= n; i++)
            sum[i] = nums[i-1] + sum[i-1];
        int minSum = 0, res = INT_MIN;
        for (int i = 1; i <= n; i++) {
            // 先更新
            res = max(res, sum[i] - minSum);
            // 再维护
            minSum = min(minSum, sum[i]);
        }
        return res;
    }
};
```

> 如果使用动态规划思想，假设dp[i]代表到i位置中最大子数组的和的话
那么如何从dp[i-1]推至dp[i]呢？dp[i]是包括nums[i]的最大连续子数组的和，
如果nums[i]是最大子数组的一员，那么该子数组和就是dp[i-1]+nums[i]
如果不是，那就让他自成一派（重新计算最大子数组）
OK，动态规划思路完成
> 

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n, 0);
        dp[0] = nums[0];
        for (int i = 1; i < n; i++)
            // 自成一派 还是 加入前辈
            dp[i] = max(nums[i], dp[i-1] + nums[i]);
        return *max_element(dp.begin(), dp.end());
    }
};
```

### **力扣1143. 最长公共子序列**

给定两个字符串 `text1` 和 `text2`，返回这两个字符串的最长 **公共子序列** 的长度。如果不存在 **公共子序列** ，返回 `0` 。

一个字符串的 **子序列** **是指这样一个新的字符串：它是由原字符串在不改变字符的相对顺序的情况下删除某些字符（也可以不删除任何字符）后组成的新字符串。

- 例如，`"ace"` 是 `"abcde"` 的子序列，但 `"aec"` 不是 `"abcde"` 的子序列。

两个字符串的 **公共子序列** 是这两个字符串所共同拥有的子序列。

> 对于LCS，可以用两个指针的笨办法
如果两个指针指向字符相同，自然是LCS
如果不同，两个指针分别前进
不对，这样会漏掉某些字符，使得最终得到的结果错误
等等，这个想法可以应用到动态规划（对每个字符的子问题）
如果两字符相等，加一寻找下一个字符即可
如果不等，我们可以假设三种情况：
1.字符串1的字符是LCS
2.字符串2的字符是LCS
3.两字符串的字符都不是LCS
取这三种情况的最大值即可
这样便可初步DP
> 

```cpp
class Solution {
public:
    int longestCommonSubsequence(string s, string t) {
        return dp(s, 0, t, 0);
    }
private:
    int dp(string s, int i, string t, int j) {
        if (i == s.length()) return 0;
        if (j == t.length()) return 0;
        // 相等
        if (s[i] == t[j])
            return 1 + dp(s, i+1, t, j+1);
        else
            return max3(
                // 假设t[j]位于LCS中
                dp(s, i, t, j+1),
                // 假设s[i]位于LCS中
                dp(s, i+1, t, j),
                // 假设两都不在LCS
                dp(s, i+1, t, j+1)
            );
    }
    int max3(int a, int b, int c) {
        return max(a, max(b, c));
    }
};
```

> 不行，初步DP会超时，那得使用记忆化优化
这里把情况三给优化掉了，因为情况三得到的字符必然是小于情况二和情况一的
> 

```cpp
class Solution {
private:
    vector<vector<int>> lt;
public:
    int longestCommonSubsequence(string s, string t) {
        int n = s.size(), m = t.size();
        lt = vector<vector<int>>(n, vector<int>(m, -1));
        return dp(s, 0, t, 0);
    }
private:
    int dp(string& s, int i, string& t, int j) {
        if (i == s.length()) return 0;
        if (j == t.length()) return 0;
        if (lt[i][j] != -1) return lt[i][j];
        // 相等
        if (s[i] == t[j])
            lt[i][j] = 1 + dp(s, i+1, t, j+1);
        else
            lt[i][j] = max(
                // 假设t[j]位于LCS中
                dp(s, i, t, j+1),
                // 假设s[i]位于LCS中
                dp(s, i+1, t, j)
            );
        return lt[i][j];
    }
};
```

> 既然都记忆化了，何不直接上二维DP
即可以从lt[i][j-1]（忽略t[j]）和lt[i-1][j]（忽略s[i]）中最大值来填补lt[i][j]
> 

```cpp
class Solution {
public:
    int longestCommonSubsequence(string s, string t) {
        int n = s.size(), m = t.size();
        vector<vector<int>> lt(n+1, vector<int>(m+1, 0));
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++)
                if (s[i-1] == t[j-1])
                    lt[i][j] = lt[i-1][j-1] + 1;
                else
                    lt[i][j] = max(lt[i-1][j], lt[i][j-1]);
        return lt[n][m];
    }
};
```

### 力扣**712. 两个字符串的最小ASCII删除和**

给定两个字符串`s1` 和 `s2`，返回 *使两个字符串相等所需删除字符的 **ASCII** 值的最小和* 。

> 二维DP直接秒杀，注意初始化即可
> 

```cpp
class Solution {
public:
    int minimumDeleteSum(string s, string t) {
        int n = s.size(), m = t.size();
        vector<vector<int>> lt(n+1, vector<int>(m+1, 0));
        for (int i = 1; i <= n; i++) lt[i][0] = lt[i-1][0] + s[i-1];

                if (s[i-1] == t[j-1])
                    lt[i][j] = lt[i-1][j-1];
                else
                    lt[i][j] = min(
                        lt[i-1][j] + s[i-1],
                        lt[i][j-1] + t[j-1]
                    );
        return lt[n][m];
    }
};
```
