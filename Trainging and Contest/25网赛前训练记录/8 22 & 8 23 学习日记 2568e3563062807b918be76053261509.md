## C1. The Cunning Seller (easy version)

This is the easy version of the problem. The easy version differs from the hard one in that it requires determining the minimum cost with the least number of deals, while the hard version requires determining the minimum cost with a limited number of deals.

After the cunning seller sold three watermelons instead of one, he decided to increase his profit — namely, he bought even more watermelons. Now he can sell $3^x$ watermelons for $3^{x+1} + x \cdot 3^{x-1}$ coins, where $x$ is a non-negative integer. Such a sale is called a deal.

A calculating buyer came to him, but he has critically little time. Because of this, he wants to buy exactly $n$ watermelons, making the **least** possible number of deals.

The buyer is in a hurry and has therefore turned to you to determine the minimum number of coins he must pay the seller for $n$ watermelons, considering that he will make the **least** possible number of deals.

> 【补题】重点是题目给与的 $n$ 是任意的，非3的幂，即不能推断n是3的多少次方来解题。每次可以以 $3^{x+1} + x * 3^{x-1}$ 的硬币买 $3^x$ 的西瓜，$x$ 是非负整数，即也不能通过 $log_{3} n$ 来求得 $x$ 。$n$必定是多种套餐的结合体。这样看，必定存在一种贪心做法，即每次都选最靠近的交易套餐，例如对于8个西瓜，则会先选两个 $3^1$ 即 x = 1 这个套餐，剩下2个，则可以使用2个 $3^0$ 即 $x = 0$ 这个套餐，若设 $f(x) = 3^{x+1} + x * 3^{x-1}$ ，则当 n = 8 时其花费就是 $2*f(1)+2*f(0)$。
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
vector<long long> cost;
long long int solve() {
    long long n;cin>>n;
    long long sz = 1;
    long long min_cost = 0;
    // 对于任何n
    // 都是每一位上的三进制乘以f(i)
    while (n) {
        min_cost += (n % 3) * cost[sz];
        n /= 3;
        sz++;
    }
    return min_cost;
}
int main() {

        cnt *= 3;
    }

    }
```

## 力扣547.省份数量【复训】

有 `n` 个城市，其中一些彼此相连，另一些没有相连。如果城市 `a` 与城市 `b` 直接相连，且城市 `b` 与城市 `c` 直接相连，那么城市 `a` 与城市 `c` 间接相连。

**省份** 是一组直接或间接相连的城市，组内不含其他没有相连的城市。

给你一个 `n x n` 的矩阵 `isConnected` ，其中 `isConnected[i][j] = 1` 表示第 `i` 个城市和第 `j` 个城市直接相连，而 `isConnected[i][j] = 0` 表示二者不直接相连。

返回矩阵中 **省份** 的数量。

> 并查集版本
>

```cpp
class Solution {
    public:
        class UnionFold {
            private:
                vector<int> parent;
                vector<int> size;
                int count;
            public:
                UnionFold(int n) : parent(n), size(n, 1), count(n) {
                    for (int i = 0; i < n; i++)
                        parent[i] = i;
                }
                bool merge(int p, int q) {
                    int rootP = find(p);
                    int rootQ = find(q);
                    if (rootP == rootQ) return false;

                    if (size[rootP] > size[rootQ]) {
                        parent[rootQ] = rootP;
                        size[rootP] += size[rootQ];
                    } else {
                        parent[rootP] = rootQ;
                        size[rootQ] += size[rootP];
                    }
                    count--;
                    return true;
                }
                int find(int x) {
                    while (x != parent[x]) {
                        parent[x] = parent[parent[x]];
                        x = parent[x];
                    }
                    return x;
                }
                int Count() { return count; }
        };
        int findCircleNum(vector<vector<int>>& isConnected) {
            int n = isConnected.size();
            UnionFold uf(n);
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++) {
                    if (isConnected[i][j] == 1)
                        uf.merge(i, j);
                }
            return uf.Count();
        }
    };
```

## 力扣1584.连接所有点的最小费用【复训】

给你一个`points` 数组，表示 2D 平面上的一些点，其中 `points[i] = [xi, yi]` 。

连接点 `[xi, yi]` 和点 `[xj, yj]` 的费用为它们之间的 **曼哈顿距离** ：`|xi - xj| + |yi - yj|` ，其中 `|val|` 表示 `val` 的绝对值。

请你返回将所有点连接的最小总费用。只有任意两点之间 **有且仅有** 一条简单路径时，才认为所有点都已连接。

> Prim算法
> 

```cpp
class Solution {
private:
    vector<bool> flag;
    vector<vector<pair<int,int>>> mp;
    int prim(int start) {
        priority_queue<tuple<int,int,int>,vector<tuple<int,int,int>>,greater<tuple<int,int,int>>> q;

        int sum = 0;
        q.emplace(0, 0, start);

        while(!q.empty()) {
            auto t = q.top();q.pop();
            int w = get<0>(t), u = get<1>(t), v = get<2>(t);
            if (flag[v]) continue;
            flag[v] = true;
            sum += w;
            for (pair<int,int>& p : mp[v]) {
                int y = p.first, w2 = p.second;
                if (!flag[y]) q.emplace(w2, v, y);
            }
        }

        if (find(flag.begin(), flag.end(), false) != flag.end()) return 0;
        return sum;
    }
public:
    int minCostConnectPoints(vector<vector<int>>& points) {
        int n = points.size();
        flag = vector<bool>(n, false);
        mp = vector<vector<pair<int,int>>>(n);
        for (int i = 0; i < n; i++)
            for (int j = i + 1; j < n; j++) {
                int x1 = points[i][0], y1 = points[i][1];
                int x2 = points[j][0], y2 = points[j][1];
                int val = abs(x1 - x2) + abs(y1 - y2);
                mp[i].push_back({j, val});
                mp[j].push_back({i, val});
            }
        return prim(0);
    }
};
```

```cpp
    private:
        struct Edge {
            int u, v, w;
            Edge(int a, int b, int c) : u(a), v(b), w(c) {}
        };
        vector<Edge> mp;
        class UnionFold {
            private:
                vector<int> parent;
                vector<int> size;
                int count;
            public:
                UnionFold(int n) : parent(n), size(n, 1), count(n) {
                    for (int i = 0; i < n; i++)
                        parent[i] = i;
                }
                bool merge(int p, int q) {
                    int rootP = find(p);
                    int rootQ = find(q);
                    if (rootP == rootQ) return false;

                    if (size[rootP] > size[rootQ]) {
                        parent[rootQ] = rootP;
                        size[rootP] += size[rootQ];
                    } else {
                        parent[rootP] = rootQ;
                        size[rootQ] += size[rootP];
                    }
                    count--;
                    return true;
                }
                int find(int x) {
                    while (x != parent[x]) {
                        parent[x] = parent[parent[x]];
                        x = parent[x];
                    }
                    return x;
                }
                int Count() { return count; }
        };
        int Kruskal(int n) {
            sort(mp.begin(), mp.end(), [](Edge& a, Edge& b){
                return a.w < b.w;
            });
            int sum = 0, cnt = 0;
            UnionFold uf(n);
            for (Edge& t : mp) {
                int u = t.u, v = t.v, w = t.w;
                if(uf.merge(u, v)) {
                    sum += w;
                    cnt++;
                    if (cnt == n-1) break;
                }
            }
            return cnt < n-1 ? 0 : sum;
        }
public:
    int minCostConnectPoints(vector<vector<int>>& points) {
        int n = points.size();
        for (int i = 0; i < n; i++)
            for (int j = i + 1; j < n; j++) {
                int x1 = points[i][0], y1 = points[i][1];
                int x2 = points[j][0], y2 = points[j][1];
                int val = abs(x1 - x2) + abs(y1 - y2);
                mp.push_back(Edge(i, j, val));
                mp.push_back(Edge(j, i, val));
            }
        return Kruskal(n);
    }
};
```

## 力扣1631.最小体力消耗路径【复训】

你准备参加一场远足活动。给你一个二维 `rows x columns` 的地图 `heights` ，其中 `heights[row][col]` 表示格子 `(row, col)` 的高度。一开始你在最左上角的格子 `(0, 0)` ，且你希望去最右下角的格子 `(rows-1, columns-1)` （注意下标从 **0** 开始编号）。你每次可以往 **上**，**下**，**左**，**右** 四个方向之一移动，你想要找到耗费 **体力** 最小的一条路径。

一条路径耗费的 **体力值** 是路径上相邻格子之间 **高度差绝对值** 的 **最大值** 决定的。

请你返回从左上角走到右下角的最小 **体力消耗值** 。

> Dijistar
> 

```cpp
class Solution {
private:
    struct State {
        int start, node, dis;
        State(int s, int n, int d) : start(s), node(n), dis(d) {}
        bool operator<(const State& other) const {
            return dis > other.dis;
        }
    };
    int Dijistar(vector<vector<int>>& martix) {
        int n = martix.size(), m = martix[0].size();
        priority_queue<State> q;
        vector<int> dx = {-1, 0, 1, 0}, dy = {0, 1, 0, -1};
        // 错误1::数组范围过大
        vector<vector<int>> dis(n, vector<int>(m, -1));
        q.emplace(0, 0, 0);
        while (!q.empty()) {
            State t = q.top();q.pop();
            int x = t.start, y = t.node, w = t.dis;
            if (dis[x][y] != -1) continue;
            dis[x][y] = w;
            // 错误2:返回时机错误
            if (x == n-1 && y == m-1) return w;
            for (int i = 0; i < 4; i++) {
                int a = x + dx[i], b = y + dy[i];
                if (a < 0 || b < 0 || a >= n || b >= m) continue;
                int nw = max(w, abs(martix[x][y] - martix[a][b]));
                if (dis[a][b] != -1) continue;
                q.emplace(a, b, nw);
            }
        }
        return -1;
    }
public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        return Dijistar(heights);
    }
};
```

```cpp
class Solution {
private:
    vector<bool> flag;
    void dfs(vector<int>& nums, vector<vector<int>>& res, deque<int>& path) {
        if (path.size() == nums.size()) {
            res.push_back(vector<int>(path.begin(),path.end()));
            return;
        }
        for (int i = 0; i < nums.size(); i++) {
            if (flag[i]) continue;

            flag[i] = false;
            path.pop_back();
        }

        flag = vector<bool>(n+1, false);
        vector<vector<int>> res;
        deque<int> path;
        dfs(nums, res, path);
        return res;
    }
};
```

```cpp
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == NULL) return NULL;
        if (root->val == p->val || root->val == q->val) return root;
        TreeNode* left = lowestCommonAncestor(root->left, p, q);
        TreeNode* right = lowestCommonAncestor(root->right, p, q);
        if (right && left) return root;
        return left?left:right;
    }
};
```

```cpp
public:
    bool findSafeWalk(vector<vector<int>>& grid, int health) {
        int n = grid.size(), m = grid[0].size();
        vector<vector<int>> dis(n, vector<int>(m, INT_MAX));
        dis[0][0] = grid[0][0];
        deque<pair<int,int>> q;

                    dis[x][y] = dis[i][j] + cost;
```

```cpp
class Solution {
private:
    struct State {
        int node;
        double dis;
        State(int n, double d) : node(n), dis(d) {}
        // [大根堆]题目要概率越大越好
        bool operator<(const State& other) const {
            return dis < other.dis;
        }
    };
public:
    double maxProbability(int n, vector<vector<int>>& edges, vector<double>& succProb, int start, int end) {

        for (int i = 0; i < edges.size(); i++) {

            if (u == end) return d;
            if (dis[u] != 0) continue;
            dis[u] = d;
            for (auto& p : mp[u]) {
                int v = p.first;
                double w = d * p.second;
                if (dis[v] == 0) q.emplace(v, w);
            }
        }
```

Polycarp is designing a level for a game. The level consists of $n$ segments on the number line, where the $i$\-th segment starts at the point with coordinate $l_i$ and ends at the point with coordinate $r_i$.

The player starts the level at the point with coordinate $0$. In one move, they can move to any point that is within a distance of no more than $k$. After their $i$\-th move, the player must land within the $i$\-th segment, that is, at a coordinate $x$ such that $l_i \le x \le r_i$. This means:

- After the first move, they must be inside the first segment (from $l_1$ to $r_1$);
- After the second move, they must be inside the second segment (from $l_2$ to $r_2$);
- ...
- After the $n$\-th move, they must be inside the $n$\-th segment (from $l_n$ to $r_n$).

The level is considered completed if the player reaches the $n$\-th segment, following the rules described above. After some thought, Polycarp realized that it is impossible to complete the level with some values of $k$.

Polycarp does not want the level to be too easy, so he asks you to determine the minimum integer $k$ with which it is possible to complete the level.

> 【补题】
从0开始，有多个线段，请问能否找到一个最小值使得每次都能到达一个区间？
就是存在一个值，使得每一个区间之间都有交集，这样便可从第一个区间到最后一个区间。
最小值直接上二分。然后动态更新区间（即更新交集），只要使得交集存在即有解。
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
vector<pair<int,int>> mp;
bool check(int mid) {
    int l = 0, r = 0; 
    for (pair<int,int>& p : mp) {
        int l1 = p.first, r1 = p.second;
        l = max(l - mid, l1);
        r = min(r + mid, r1);
        if (l > r) return false;
    }
    return true;
}
int solve(){
    int n;cin>>n;
    mp.clear();
    int m = 0;
    for (int i = 0; i < n; i++) {
        int a, b;cin>>a>>b;
        mp.push_back({a, b});
        m = max(m, b);
    }
    int left = 0, right = m;
    while (right > left) {
        int mid = left + ((right - left) >> 1);
        if (check(mid)) right = mid;
				else left = mid + 1;
    }
    return left;
}
int main() {
    IOS;
    int t;cin>>t;
    while (t--) {
        cout<<solve()<<"\n";
    }
    return 0;
}
```

## E. Good Triples

Given a non-negative integer number $n$ ($n \ge 0$). Let's say a triple of non-negative integers $(a, b, c)$ is good if $a + b + c = n$, and $digsum(a) + digsum(b) + digsum(c) = digsum(n)$, where $digsum(x)$ is the sum of digits of number $x$.

For example, if $n = 26$, then the pair $(4, 12, 10)$ is good, because $4 + 12 + 10 = 26$, and $(4) + (1 + 2) + (1 + 0) = (2 + 6)$.

Your task is to find the number of good triples for the given number $n$. The order of the numbers in a triple matters. For example, the triples $(4, 12, 10)$ and $(10, 12, 4)$ are two different triples.

> 【补题】
> 
> 
> 题目要求对于n找三个数，这三个数不仅和等于n，且数位之和也要等于n的数位之和。那么就是不能进位（比赛时想到了），我们取出n的每一位，对n的每一位d，找到有多少对a+b+c=d，按概率相乘即可找到答案。
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
vector<pair<int,int>> mp;
int solve(){
    int n;cin>>n;
    int cnt = 0;
    while (n > 0) {
        int d = n % 10;
        n /= 10;
        int c = 0;
        for (int i = 0; i < d + 1; i++)
            for (int j = 0; j < d + 1; j++) 
                if (d - i - j >= 0)
                    c += 1;
        cnt *= c;
    }
    return cnt;
}
int main() {
    IOS;
    int t;cin>>t;
    while (t--) {
        cout<<solve()<<"\n";
    }
    return 0;
}
```

## F. Shift and Reverse

Given an array of integers $a_1, a_2, \ldots, a_n$. You can make two types of operations with this array:

- Shift: move the last element of array to the first place, and shift all other elements to the right, so you get the array $a_n, a_1, a_2, \ldots, a_{n-1}$.
- Reverse: reverse the whole array, so you get the array $a_n, a_{n-1}, \ldots, a_1$.

Your task is to sort the array in non-decreasing order using the minimal number of operations, or say that it is impossible.

> 【补题】
> 
> 
> 比赛时还在模拟，模拟翻转和移动。
> 
> 先将数组翻转再复制，虽多余但在复杂情况下优势最大。
> 然后在数组中如果存在一个升序或降序序列，代表可以通过移动、翻转来得到题目要求的升序序列，否则没有。
> 假设复制后的数组中存在一段有序序列[L,R]，
> 此时如果是升序序列，即代表在原数组中是降序，需要考虑L-1和m-R+1谁最小；
> 如果是降序序列，即代表在原数组中是升序序列，需要考虑L-1+1和m-R谁最小。
> 

```cpp
#include<bits/stdc++.h>
using namespace std;
#define IOS ios::sync_with_stdio(false);cin.tie(0);
int solve(){
    int n;cin>>n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin>>a[i];
    if (n == 1) return 0;
    reverse(a.begin(), a.end());
    a.insert(a.end(), a.begin(), a.end());
    int m = a.size();
    vector<int> p(m, 0), q(m, 0);
    for (int i = 0; i < m - 1; i++) {
        p[i + 1] = (a[i] >= a[i + 1]) ? p[i] + 1 : 0;
        q[i + 1] = (a[i] <= a[i + 1]) ? q[i] + 1 : 0;
    }
    int res = INT_MAX;
    for (int i = n - 1; i < m; i++) { 
        if (p[i] == n - 1) res = min(res, min(i - n + 1, m - i + 1));
        if (q[i] == n - 1) res = min(res, min(m - i, i - n + 2));
    }
    return res == INT_MAX ? -1 : res;
}
int main() {
    IOS;
    int t;cin>>t;
    while (t--) {
        cout<<solve()<<"\n";
    }
    return 0;
}
```
