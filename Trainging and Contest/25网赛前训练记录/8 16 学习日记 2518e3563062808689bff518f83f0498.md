力扣144.二叉树的前序遍历

给你二叉树的根节点 `root` ，返回它节点值的 **前序** **遍历。
普通版本

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void per(TreeNode* root,vector<int>& res){
        if(root == NULL) return;
        res.push_back(root->val);
        per(root->left,res);
        per(root->right,res);
    }
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> res;
        per(root,res);
        return res;
    }
};
```

栈版本

```cpp
class Solution {
public:
    void per(TreeNode* root,stack<int>& res){
        if(root == NULL) return;
        res.push(root->val);
        per(root->left,res);
        per(root->right,res);
    }
    vector<int> preorderTraversal(TreeNode* root) {
        stack<int> ans;
        per(root,ans);
        vector<int> res;
        while(!ans.empty()){
            res.push_back(ans.top());
            ans.pop();
        }
        reverse(res.begin(),res.end());
        return res;
    }
};
```

力扣102. 二叉树的层序遍历

给你二叉树的根节点 `root` ，返回其节点值的 **层序遍历** 。 （即逐层地，从左到右访问所有节点）。

层序是根据每层顺序，从上到下，从左到右访问所有的节点。那么可以使用队列，通过先行存储根节点，在循环中先提取根节点的值，存储他的孩子节点至队列，然后将该节点的值封存作为一层，接下来再访问他的子节点，即下一层，这样便做到了层序遍历。

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        queue<TreeNode*> q;
        vector<vector<int>> res;
        if(root == NULL) return res;

        //先行存储根节点
        q.push(root);

        while(!q.empty()){
            //取出当前长度[即该层长度]
            int n=q.size();
            vector<int> t;

            //循环当前层的节点值
            for(int i=0;i<n;i++){
                TreeNode* temp=q.front();
                //保存当前节点的值
                t.push_back(temp->val);

                if(temp->left!=NULL) q.push(temp->left);
                if(temp->right!=NULL) q.push(temp->right);
                //将该节点从队首弹出
                q.pop();
                //注:因为i是从0到n-1循环的,此时弹出不用更新n的值
            }
            res.push_back(t);
        }
        return res;
    }
};
```

力扣104.二叉树的最大深度

给定一个二叉树 `root` ，返回其最大深度。

二叉树的 **最大深度** 是指从根节点到最远叶子节点的最长路径上的节点数。

遍历，寻找最大深度，当前为空作为递归终点。

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int deep(TreeNode* root){
        if(root == NULL) return 0;
        return 1 + max(deep(root->left),deep(root->right));
    }
    int maxDepth(TreeNode* root) {
        return deep(root);
    }
};
```

超级简化版

```cpp
class Solution {
public:
    int maxDepth(TreeNode* root) {
        return root==NULL?0:(1+max(maxDepth(root->left),maxDepth(root->right)));
    }
};
```

101. 对称二叉树

给你一个二叉树的根节点 `root` ， 检查它是否轴对称。

```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool rotate(TreeNode* l,TreeNode* r){
        // [递归出口1]两边都空,对称
        if(l==NULL&&r==NULL) return true;
        // [递归出口2]一边为空另一边不空,不对称
        if(l==NULL||r==NULL) return false;
        // 错误注意:优先判断是否为空,否则会报错
        // [递归出口3]值不同,不对称
        if(l->val!=r->val) return false;
        // 左节点的左孩子和右节点的右孩子
        // 左节点的右孩子和右节点的左孩子
        // 这两个相等才是对称
        return rotate(l->left,r->right)&&rotate(l->right,r->left);
    }
    bool isSymmetric(TreeNode* root) {
        if(root == NULL) return true;
        return rotate(root->left,root->right);
    }
};

```
