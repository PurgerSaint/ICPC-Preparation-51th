# 力扣1143.最长公共子序列
> 初版DFS，直接超时:  
```cpp
class Solution {
public:
    int longestCommonSubsequence(string s, string t) {
        return dp(0, 0, s, t);
    }
private:
    int dp(int i, int j, const string& s, const string& t) {
        int n = s.length(), m = t.length();
        if (i < 0 || j < 0 || i >= n || j >= m) return 0;
        if (s[i] == t[j]) return 1 + dp(i + 1, j + 1, s, t);
        return max(dp(i + 1, j, s, t), dp(i, j + 1, s, t));
    }
};
```
做了初版记忆化，经*GPT*指导下，我的DFS返回错误  
由于我给数组初始化均为0，设定`!memory[i][j]`时返回，直接在第一个元素返回了  
应给数组初始化为-1，代表未处理
```cpp
class Solution {
private:
    int n, m;
    vector<vector<int>> memory;
public:
    int longestCommonSubsequence(string s, string t) {
        n = s.length(), m = t.length();
        memory = vector<vector<int>>(n, vector<int>(m, -1));
        return dp(0, 0, s, t);
    }
private:
    int dp(int i, int j, const string& s, const string& t) {
        if (i < 0 || j < 0 || i >= n || j >= m) return 0;
        if (memory[i][j] != -1) return memory[i][j];
        if (s[i] == t[j]) return memory[i][j] = 1 + dp(i + 1, j + 1, s, t);
        return memory[i][j] = max(dp(i + 1, j, s, t), dp(i, j + 1, s, t));
    }
};
```

经*GPT*提醒下，应该是直接拿取或不拿取当前物品

```cpp
int t, m;
vector<pair<int, int>> b;
vector<vector<int>> memory;
int dfs(int i, int j) {
    if (i < 0 || j < 0 || i >= m || j >= t) return 0;
    if (memory[i][j] != -1) return memory[i][j];
    int ans = dfs(i + 1, j);
    auto& [time, cost] = b[i];
    if (j + time <= t)
        ans = max(ans, cost + dfs(i + 1, j + time));
    return memory[i][j] = ans;
}
```
